### 1.2 核心组件

#### 路由管理器 (Router Manager)
- **功能**：智能路由请求到合适的n8n工作流
- **职责**：
  - 分析用户输入内容和意图
  - 根据业务规则选择目标工作流
  - 管理工作流之间的依赖关系
  - 处理工作流链式调用

#### 状态管理器 (State Manager)
- **功能**：统一管理所有工作流的执行状态
- **职责**：
  - 跟踪工作流执行进度
  - 管理工作流生命周期
  - 提供状态查询接口
  - 处理状态变更通知

#### 通信管理器 (Communication Manager)
- **功能**：管理与n8n工作流的通信
- **职责**：
  - HTTP请求的发送和响应处理
  - WebSocket连接管理
  - 错误重试和容错处理
  - 响应数据格式化

## 2. 工作流分析

```
# backend/open_webui/services/workflow_router.py
import re
import json
from typing import Dict, Any, Optional, List
from enum import Enum
from dataclasses import dataclass

class WorkflowType(Enum):
    MAIN_WORKFLOW = "main_workflow"
    COMPANY_INFO = "company_info"
    VIRAL_LEARNING = "viral_learning"
    VIDEO_SCRAPING = "video_scraping"

@dataclass
class RoutingRule:
    pattern: str
    workflow_type: WorkflowType
    priority: int
    conditions: List[str] = None

class WorkflowRouter:
    def __init__(self):
        self.routing_rules = self._initialize_routing_rules()
        self.workflow_endpoints = {
            WorkflowType.MAIN_WORKFLOW: "https://webhook-n8n.hsai.cc/webhook/main-workflow",
            WorkflowType.COMPANY_INFO: "https://webhook-n8n.hsai.cc/webhook/company-info",
            WorkflowType.VIRAL_LEARNING: "https://webhook-n8n.hsai.cc/webhook/viral-learning",
            WorkflowType.VIDEO_SCRAPING: "https://webhook-n8n.hsai.cc/webhook/video-scraping"
        }
    
    def _initialize_routing_rules(self) -> List[RoutingRule]:
        return [
            # 企业信息收集规则
            RoutingRule(
                pattern=r"(企业|公司|组织).*(信息|资料|分析)",
                workflow_type=WorkflowType.COMPANY_INFO,
                priority=10,
                conditions=["has_file_upload"]
            ),
            RoutingRule(
                pattern=r"(上传|解析|分析).*(文件|文档|资料)",
                workflow_type=WorkflowType.COMPANY_INFO,
                priority=9
            ),
            
            # 视频相关规则
            RoutingRule(
                pattern=r"(视频|爬取|抓取).*(分析|关键词|内容)",
                workflow_type=WorkflowType.VIDEO_SCRAPING,
                priority=8
            ),
            RoutingRule(
                pattern=r"(抖音|快手|小红书|B站).*(视频|内容)",
                workflow_type=WorkflowType.VIDEO_SCRAPING,
                priority=8
            ),
            
            # 爆款学习规则
            RoutingRule(
                pattern=r"(爆款|热门|趋势).*(学习|分析|研究)",
                workflow_type=WorkflowType.VIRAL_LEARNING,
                priority=7
            ),
            
            # 默认主工作流
            RoutingRule(
                pattern=r".*",
                workflow_type=WorkflowType.MAIN_WORKFLOW,
                priority=1
            )
        ]
    
    def route_request(self, user_input: str, context: Dict[str, Any] = None) -> WorkflowType:
        """根据用户输入路由到合适的工作流"""
        context = context or {}
        
        # 按优先级排序规则
        sorted_rules = sorted(self.routing_rules, key=lambda x: x.priority, reverse=True)
        
        for rule in sorted_rules:
            if self._match_rule(rule, user_input, context):
                return rule.workflow_type
        
        # 默认返回主工作流
        return WorkflowType.MAIN_WORKFLOW
    
    def _match_rule(self, rule: RoutingRule, user_input: str, context: Dict[str, Any]) -> bool:
        # 检查模式匹配
        if not re.search(rule.pattern, user_input, re.IGNORECASE):
            return False
        
        # 检查条件
        if rule.conditions:
            for condition in rule.conditions:
                if not self._check_condition(condition, context):
                    return False
        
        return True
    
    def _check_condition(self, condition: str, context: Dict[str, Any]) -> bool:
        if condition == "has_file_upload":
            return context.get("has_files", False)
        # 可以添加更多条件检查
        return True
    
    def get_workflow_endpoint(self, workflow_type: WorkflowType) -> str:
        return self.workflow_endpoints.get(workflow_type)
```

### 3.2 状态管理器实现

```
# backend/open_webui/services/workflow_state_manager.py
from typing import Dict, Any, Optional, List, Callable
from enum import Enum
from datetime import datetime, timedelta
import asyncio
import uuid

class WorkflowStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

class WorkflowStateManager:
    def __init__(self):
        self.active_workflows: Dict[str, Dict[str, Any]] = {}
        self.workflow_history: List[Dict[str, Any]] = []
        self.status_callbacks: Dict[str, List[Callable]] = {}
        self.cleanup_interval = 3600  # 1小时清理一次
        
    async def create_workflow_session(self, 
                                    workflow_type: str,
                                    user_id: str,
                                    initial_data: Dict[str, Any]) -> str:
        """创建工作流会话"""
        session_id = str(uuid.uuid4())
        
        workflow_session = {
            "session_id": session_id,
            "workflow_type": workflow_type,
            "user_id": user_id,
            "status": WorkflowStatus.PENDING,
            "created_at": datetime.now(),
            "updated_at": datetime.now(),
            "initial_data": initial_data,
            "execution_data": {},
            "error_info": None,
            "progress": 0.0
        }
        
        self.active_workflows[session_id] = workflow_session
        await self._notify_status_change(session_id, WorkflowStatus.PENDING)
        
        return session_id
    
    async def update_workflow_status(self, 
                                   session_id: str, 
                                   status: WorkflowStatus,
                                   execution_data: Dict[str, Any] = None,
                                   error_info: str = None,
                                   progress: float = None):
        """更新工作流状态"""
        if session_id not in self.active_workflows:
            raise ValueError(f"Workflow session {session_id} not found")
        
        workflow = self.active_workflows[session_id]
        workflow["status"] = status
        workflow["updated_at"] = datetime.now()
        
        if execution_data:
            workflow["execution_data"].update(execution_data)
        
        if error_info:
            workflow["error_info"] = error_info
        
        if progress is not None:
            workflow["progress"] = progress
        
        await self._notify_status_change(session_id, status)
        
        # 如果工作流完成或失败，移动到历史记录
        if status in [WorkflowStatus.COMPLETED, WorkflowStatus.FAILED, WorkflowStatus.CANCELLED]:
            await self._archive_workflow(session_id)
    
    def get_workflow_status(self, session_id: str) -> Optional[Dict[str, Any]]:
        """获取工作流状态"""
        return self.active_workflows.get(session_id)
    
    def get_user_workflows(self, user_id: str) -> List[Dict[str, Any]]:
        """获取用户的所有工作流"""
        user_workflows = []
        
        # 活跃工作流
        for workflow in self.active_workflows.values():
            if workflow["user_id"] == user_id:
                user_workflows.append(workflow)
        
        # 历史工作流（最近的10个）
        recent_history = [w for w in self.workflow_history if w["user_id"] == user_id][-10:]
        user_workflows.extend(recent_history)
        
        return sorted(user_workflows, key=lambda x: x["updated_at"], reverse=True)
    
    def subscribe_to_status_changes(self, session_id: str, callback: Callable):
        """订阅状态变更通知"""
        if session_id not in self.status_callbacks:
            self.status_callbacks[session_id] = []
        
        self.status_callbacks[session_id].append(callback)
    
    def unsubscribe_from_status_changes(self, session_id: str, callback: Callable):
        """取消订阅状态变更通知"""
        if session_id in self.status_callbacks:
            try:
                self.status_callbacks[session_id].remove(callback)
            except ValueError:
                pass
    
    async def _notify_status_change(self, session_id: str, status: WorkflowStatus):
        """通知状态变更"""
        if session_id in self.status_callbacks:
            workflow_data = self.active_workflows.get(session_id)
            
            for callback in self.status_callbacks[session_id]:
                try:
                    await callback(session_id, status, workflow_data)
                except Exception as e:
                    print(f"Error in status callback: {e}")
    
    async def _archive_workflow(self, session_id: str):
        """归档工作流到历史记录"""
        if session_id in self.active_workflows:
            workflow = self.active_workflows.pop(session_id)
            self.workflow_history.append(workflow)
            
            # 清理回调
            if session_id in self.status_callbacks:
                del self.status_callbacks[session_id]
            
            # 限制历史记录数量
            if len(self.workflow_history) > 1000:
                self.workflow_history = self.workflow_history[-1000:]
    
    async def cleanup_expired_workflows(self):
        """清理过期的工作流"""
        current_time = datetime.now()
        expired_sessions = []
        
        for session_id, workflow in self.active_workflows.items():
            # 超过24小时的工作流视为过期
            if current_time - workflow["updated_at"] > timedelta(hours=24):
                expired_sessions.append(session_id)
        
        for session_id in expired_sessions:
            await self.update_workflow_status(session_id, WorkflowStatus.CANCELLED)
```

```
# backend/open_webui/services/workflow_communication_manager.py
import aiohttp
import asyncio
import json
from typing import Dict, Any, Optional
from datetime import datetime
import logging

class WorkflowCommunicationManager:
    def __init__(self):
        self.session = None
        self.logger = logging.getLogger(__name__)
        self.retry_config = {
            "max_retries": 3,
            "retry_delay": 1.0,
            "backoff_factor": 2.0
        }
    
    async def __aenter__(self):
        self.session = aiohttp.ClientSession(
            timeout=aiohttp.ClientTimeout(total=300),  # 5分钟超时
            headers={
                "Content-Type": "application/json",
                "User-Agent": "OpenWebUI-WorkflowOrchestrator/1.0"
            }
        )
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session:
            await self.session.close()
    
    async def execute_workflow(self, 
                             endpoint: str, 
                             payload: Dict[str, Any],
                             session_id: str = None) -> Dict[str, Any]:
        """执行n8n工作流"""
        
        # 添加会话信息到payload
        if session_id:
            payload["session_id"] = session_id
        
        payload["timestamp"] = datetime.now().isoformat()
        
        for attempt in range(self.retry_config["max_retries"]):
            try:
                async with self.session.post(endpoint, json=payload) as response:
                    if response.status == 200:
                        result = await response.json()
                        return self._process_response(result)
                    else:
                        error_text = await response.text()
                        raise aiohttp.ClientResponseError(
                            request_info=response.request_info,
                            history=response.history,
                            status=response.status,
                            message=error_text
                        )
            
            except Exception as e:
                self.logger.warning(f"Workflow execution attempt {attempt + 1} failed: {e}")
                
                if attempt < self.retry_config["max_retries"] - 1:
                    delay = self.retry_config["retry_delay"] * (
                        self.retry_config["backoff_factor"] ** attempt
                    )
                    await asyncio.sleep(delay)
                else:
                    raise
    
    def _process_response(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """处理n8n响应数据"""
        
        # 标准化响应格式
        processed_response = {
            "success": True,
            "data": response,
            "timestamp": datetime.now().isoformat(),
            "source": "n8n_workflow"
        }
        
        # 检查响应中是否包含错误信息
        if isinstance(response, dict):
            if "error" in response or "Error" in response:
                processed_response["success"] = False
                processed_response["error"] = response.get("error") or response.get("Error")
        
        return processed_response
    
    async def health_check(self, endpoint: str) -> bool:
        """检查工作流端点健康状态"""
        try:
            async with self.session.get(f"{endpoint}/health") as response:
                return response.status == 200
        except Exception:
            return False
```

```
# backend/open_webui/services/unified_workflow_service.py
from .workflow_router import WorkflowRouter, WorkflowType
from .workflow_state_manager import WorkflowStateManager, WorkflowStatus
from .workflow_communication_manager import WorkflowCommunicationManager

class UnifiedWorkflowService:
    def __init__(self):
        self.router = WorkflowRouter()
        self.state_manager = WorkflowStateManager()
        self.communication_manager = WorkflowCommunicationManager()
    
    async def process_user_request(self, 
                                 user_input: str,
                                 user_id: str,
                                 context: Dict[str, Any] = None) -> Dict[str, Any]:
        """处理用户请求的统一入口"""
        
        try:
            # 1. 路由决策
            workflow_type = self.router.route_request(user_input, context)
            endpoint = self.router.get_workflow_endpoint(workflow_type)
            
            # 2. 创建工作流会话
            session_id = await self.state_manager.create_workflow_session(
                workflow_type.value,
                user_id,
                {"user_input": user_input, "context": context}
            )
            
            # 3. 更新状态为运行中
            await self.state_manager.update_workflow_status(
                session_id, 
                WorkflowStatus.RUNNING
            )
            
            # 4. 执行工作流
            async with self.communication_manager as comm:
                payload = self._prepare_payload(user_input, context, workflow_type)
                result = await comm.execute_workflow(endpoint, payload, session_id)
            
            # 5. 更新完成状态
            await self.state_manager.update_workflow_status(
                session_id,
                WorkflowStatus.COMPLETED if result["success"] else WorkflowStatus.FAILED,
                execution_data=result,
                error_info=result.get("error"),
                progress=100.0
            )
            
            return {
                "session_id": session_id,
                "workflow_type": workflow_type.value,
                "result": result,
                "status": "completed" if result["success"] else "failed"
            }
            
        except Exception as e:
            # 更新失败状态
            if 'session_id' in locals():
                await self.state_manager.update_workflow_status(
                    session_id,
                    WorkflowStatus.FAILED,
                    error_info=str(e)
                )
            
            raise
    
    def _prepare_payload(self, 
                        user_input: str, 
                        context: Dict[str, Any], 
                        workflow_type: WorkflowType) -> Dict[str, Any]:
        """根据工作流类型准备请求载荷"""
        
        base_payload = {
            "input": user_input,
            "context": context or {},
            "workflow_type": workflow_type.value
        }
        
        # 根据不同工作流类型定制载荷
        if workflow_type == WorkflowType.COMPANY_INFO:
            base_payload["analysis_type"] = "company_info"
            if context and context.get("files"):
                base_payload["files"] = context["files"]
        
        elif workflow_type == WorkflowType.VIDEO_SCRAPING:
            base_payload["scraping_config"] = {
                "platforms": ["douyin", "xiaohongshu"],
                "max_videos": 50
            }
        
        elif workflow_type == WorkflowType.VIRAL_LEARNING:
            base_payload["learning_config"] = {
                "analysis_depth": "deep",
                "trend_period": "7d"
            }
        
        return base_payload
    
    async def get_workflow_status(self, session_id: str) -> Optional[Dict[str, Any]]:
        """获取工作流状态"""
        return self.state_manager.get_workflow_status(session_id)
    
    async def get_user_workflows(self, user_id: str) -> List[Dict[str, Any]]:
        """获取用户工作流列表"""
        return self.state_manager.get_user_workflows(user_id)
```

```
# backend/open_webui/routers/workflow_router.py
from fastapi import APIRouter, HTTPException, BackgroundTasks
from pydantic import BaseModel
from typing import Dict, Any, Optional, List

router = APIRouter(prefix="/api/v1/workflows", tags=["workflows"])

class WorkflowRequest(BaseModel):
    input: str
    context: Optional[Dict[str, Any]] = None

class WorkflowResponse(BaseModel):
    session_id: str
    workflow_type: str
    status: str
    result: Optional[Dict[str, Any]] = None

@router.post("/execute", response_model=WorkflowResponse)
async def execute_workflow(
    request: WorkflowRequest,
    user_id: str,
    background_tasks: BackgroundTasks
):
    """执行工作流"""
    try:
        service = UnifiedWorkflowService()
        result = await service.process_user_request(
            request.input,
            user_id,
            request.context
        )
        return WorkflowResponse(**result)
    
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/status/{session_id}")
async def get_workflow_status(session_id: str):
    """获取工作流状态"""
    service = UnifiedWorkflowService()
    status = await service.get_workflow_status(session_id)
    
    if not status:
        raise HTTPException(status_code=404, detail="Workflow not found")
    
    return status

@router.get("/user/{user_id}")
async def get_user_workflows(user_id: str):
    """获取用户工作流列表"""
    service = UnifiedWorkflowService()
    workflows = await service.get_user_workflows(user_id)
    return {"workflows": workflows}
```

1. **智能路由**：根据用户输入自动选择合适的工作流
2. **状态管理**：统一管理所有工作流的执行状态
3. **通信管理**：处理与n8n的HTTP通信，包含重试和错误处理
4. **统一接口**：为OpenWebUI提供简洁的工作流调用接口
5. **扩展性**：易于添加新的工作流和路由规则

通过这个编排中心，OpenWebUI可以无缝集成现有的n8n工作流，同时为后续的功能扩展提供了良好的基础架构。
