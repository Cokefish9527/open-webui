## 1. 设计目标

构建一个完善的兼容性迁移系统，确保OpenWebUI与n8n工作流集成的平滑升级和向后兼容，支持渐进式功能部署和无缝版本迁移。

### 1.1 设计目标

构建一个完善的兼容性迁移系统，确保OpenWebUI与n8n工作流集成的平滑升级和向后兼容，支持渐进式功能部署和无缝版本迁移。

### 1.2 核心原则

- **向后兼容**：新版本完全兼容旧版本功能
- **渐进式升级**：支持分阶段功能启用
- **零停机迁移**：升级过程不影响用户使用
- **数据完整性**：确保迁移过程中数据不丢失

from dataclasses import dataclass

class VersionType(Enum):

class VersionManager:

            ),
            "1.2.0": VersionInfo(
                version="1.2.0",
                release_date="2025-10-01",
                features=["LLM响应修复", "错误处理增强"],
                breaking_changes=["API响应格式变更"],
                migration_required=True,
                compatibility_matrix={"n8n": ">=1.1.0", "llm_service": ">=2.0.0"}
            )
        }
    
    def _load_feature_flags(self) -> Dict[str, bool]:
        """加载功能开关配置"""
        return {
            "websocket_communication": False,
            "llm_response_fixer": False,
            "enhanced_error_handling": True,
            "realtime_status": False,
            "smart_ui_components": False
        }
    
    def is_compatible(self, target_version: str, current_version: str = None) -> bool:
        """检查版本兼容性"""
        current = current_version or self.current_version
        
        try:
            return semver.compare(target_version, current) >= 0
        except ValueError:
            self.logger.error(f"Invalid version format: {target_version} or {current}")
            return False
    
    def get_migration_path(self, from_version: str, to_version: str) -> List[str]:
        """获取迁移路径"""
        
        if not self.is_compatible(to_version, from_version):
            raise ValueError(f"Cannot migrate from {from_version} to {to_version}")
        
        # 获取中间版本
        intermediate_versions = []
        for version in sorted(self.supported_versions.keys(), key=semver.VersionInfo.parse):
            if (semver.compare(version, from_version) > 0 and 
                semver.compare(version, to_version) <= 0):
                intermediate_versions.append(version)
        
        return intermediate_versions
    
    def enable_feature(self, feature_name: str, enabled: bool = True) -> bool:
        """启用/禁用功能"""
        
        if feature_name not in self.feature_flags:
            self.logger.warning(f"Unknown feature: {feature_name}")
            return False
        
        self.feature_flags[feature_name] = enabled
        self._save_feature_flags()
        
        self.logger.info(f"Feature {feature_name} {'enabled' if enabled else 'disabled'}")
        return True
    
    def is_feature_enabled(self, feature_name: str) -> bool:
        """检查功能是否启用"""
        return self.feature_flags.get(feature_name, False)

# backend/open_webui/utils/feature_flags.py
import json
import os
from typing import Dict, Any, Optional, Callable
from datetime import datetime, timedelta
import logging

class FeatureFlag:
    """功能开关"""
    
    def __init__(self, name: str, enabled: bool = False, 
                 rollout_percentage: float = 0.0,
                 conditions: Dict[str, Any] = None,
                 expiry_date: datetime = None):
        self.name = name
        self.enabled = enabled
        self.rollout_percentage = rollout_percentage
        self.conditions = conditions or {}
        self.expiry_date = expiry_date
        self.created_at = datetime.now()

class FeatureFlagManager:
    """功能开关管理器"""
    
    def __init__(self, config_path: str = "feature_flags.json"):
        self.config_path = config_path
        self.flags: Dict[str, FeatureFlag] = {}
        self.user_overrides: Dict[str, Dict[str, bool]] = {}
        self.logger = logging.getLogger(__name__)
        
        self._load_flags()
    
    def _load_flags(self):
        """加载功能开关配置"""
        
        default_flags = {
            "websocket_communication": FeatureFlag(
                name="websocket_communication",
                enabled=False,
                rollout_percentage=0.0,
                conditions={"min_version": "1.1.0"}
            ),
            "llm_response_fixer": FeatureFlag(
                name="llm_response_fixer", 
                enabled=False,
                rollout_percentage=10.0,
                conditions={"user_type": "beta"}
            ),
            "enhanced_ui_components": FeatureFlag(
                name="enhanced_ui_components",
                enabled=True,
                rollout_percentage=100.0
            ),
            "realtime_status_feedback": FeatureFlag(
                name="realtime_status_feedback",
                enabled=False,
                rollout_percentage=25.0,
                conditions={"websocket_communication": True}
            )
        }
        
        if os.path.exists(self.config_path):
            try:
                with open(self.config_path, 'r') as f:
                    config = json.load(f)
                    
                for name, flag_data in config.items():
                    self.flags[name] = FeatureFlag(
                        name=name,
                        enabled=flag_data.get('enabled', False),
                        rollout_percentage=flag_data.get('rollout_percentage', 0.0),
                        conditions=flag_data.get('conditions', {}),
                        expiry_date=datetime.fromisoformat(flag_data['expiry_date']) 
                                   if flag_data.get('expiry_date') else None
                    )
            except Exception as e:
                self.logger.error(f"Failed to load feature flags: {e}")
                self.flags = default_flags
        else:
            self.flags = default_flags
            self._save_flags()
    
    def _save_flags(self):
        """保存功能开关配置"""
        
        config = {}
        for name, flag in self.flags.items():
            config[name] = {
                'enabled': flag.enabled,
                'rollout_percentage': flag.rollout_percentage,
                'conditions': flag.conditions,
                'expiry_date': flag.expiry_date.isoformat() if flag.expiry_date else None
            }
        
        try:
            with open(self.config_path, 'w') as f:
                json.dump(config, f, indent=2)
        except Exception as e:
            self.logger.error(f"Failed to save feature flags: {e}")
    
    def is_enabled(self, flag_name: str, user_context: Dict[str, Any] = None) -> bool:
        """检查功能是否启用"""
        
        if flag_name not in self.flags:
            self.logger.warning(f"Unknown feature flag: {flag_name}")
            return False
        
        flag = self.flags[flag_name]
        user_context = user_context or {}
        
        # 检查用户覆盖设置
        user_id = user_context.get('user_id')
        if user_id and user_id in self.user_overrides:
            if flag_name in self.user_overrides[user_id]:
                return self.user_overrides[user_id][flag_name]
        
        # 检查过期时间
        if flag.expiry_date and datetime.now() > flag.expiry_date:
            return False
        
        # 检查基础启用状态
        if not flag.enabled:
            return False
        
        # 检查条件
        if not self._check_conditions(flag.conditions, user_context):
            return False
        
        # 检查灰度发布百分比
        if flag.rollout_percentage < 100.0:
            return self._check_rollout(flag_name, flag.rollout_percentage, user_context)
        
        return True
    
    def _check_conditions(self, conditions: Dict[str, Any], context: Dict[str, Any]) -> bool:
        """检查启用条件"""
        
        for key, expected_value in conditions.items():
            if key == "min_version":
                current_version = context.get('version', '1.0.0')
                if not self._version_meets_requirement(current_version, expected_value):
                    return False
            
            elif key == "user_type":
                user_type = context.get('user_type', 'regular')
                if user_type != expected_value:
                    return False
            
            elif key in self.flags:
                # 依赖其他功能开关
                if not self.is_enabled(key, context):
                    return False
            
            else:
                # 直接比较上下文值
                if context.get(key) != expected_value:
                    return False
        
        return True
    
    def _check_rollout(self, flag_name: str, percentage: float, context: Dict[str, Any]) -> bool:
        """检查灰度发布"""
        
        # 基于用户ID或会话ID生成一致的哈希值
        identifier = context.get('user_id') or context.get('session_id', 'anonymous')
        hash_value = hash(f"{flag_name}:{identifier}") % 100
        
        return hash_value < percentage
    
    def set_user_override(self, user_id: str, flag_name: str, enabled: bool):
        """设置用户级别的功能开关覆盖"""
        
        if user_id not in self.user_overrides:
            self.user_overrides[user_id] = {}
        
        self.user_overrides[user_id][flag_name] = enabled
        self.logger.info(f"User override set: {user_id} - {flag_name} = {enabled}")
    
    def gradual_rollout(self, flag_name: str, target_percentage: float, 
                       step_size: float = 10.0, interval_hours: int = 24):
        """渐进式功能发布"""
        
        if flag_name not in self.flags:
            raise ValueError(f"Unknown feature flag: {flag_name}")
        
        flag = self.flags[flag_name]
        current_percentage = flag.rollout_percentage
        
        if current_percentage >= target_percentage:
            self.logger.info(f"Feature {flag_name} already at target rollout: {target_percentage}%")
            return
        
        # 计算下一步的百分比
        next_percentage = min(current_percentage + step_size, target_percentage)
        flag.rollout_percentage = next_percentage
        
        self._save_flags()
        
        self.logger.info(f"Feature {flag_name} rollout updated: {current_percentage}% -> {next_percentage}%")
        
        # 如果还没达到目标，安排下次更新
        if next_percentage < target_percentage:
            # 这里可以集成定时任务系统
            pass

# backend/open_webui/migrations/migration_manager.py
import os
import json
import sqlite3
from typing import List, Dict, Any, Optional
from datetime import datetime
import logging

class Migration:
    """数据库迁移基类"""
    
    def __init__(self, version: str, description: str):
        self.version = version
        self.description = description
        self.timestamp = datetime.now()
    
    def up(self, connection):
        """执行迁移"""
        raise NotImplementedError
    
    def down(self, connection):
        """回滚迁移"""
        raise NotImplementedError

class MigrationManager:
    """数据库迁移管理器"""
    
    def __init__(self, db_path: str, migrations_dir: str = "migrations"):
        self.db_path = db_path
        self.migrations_dir = migrations_dir
        self.logger = logging.getLogger(__name__)
        
        self._ensure_migration_table()
    
    def _ensure_migration_table(self):
        """确保迁移记录表存在"""
        
        with sqlite3.connect(self.db_path) as conn:
            conn.execute("""
                CREATE TABLE IF NOT EXISTS schema_migrations (
                    version TEXT PRIMARY KEY,
                    description TEXT,
                    applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    rollback_sql TEXT
                )
            """)
            conn.commit()
    
    def get_applied_migrations(self) -> List[str]:
        """获取已应用的迁移"""
        
        with sqlite3.connect(self.db_path) as conn:
            cursor = conn.execute("SELECT version FROM schema_migrations ORDER BY version")
            return [row[0] for row in cursor.fetchall()]
    
    def get_pending_migrations(self) -> List[Migration]:
        """获取待执行的迁移"""
        
        applied = set(self.get_applied_migrations())
        all_migrations = self._load_migrations()
        
        return [m for m in all_migrations if m.version not in applied]
    
    def _load_migrations(self) -> List[Migration]:
        """加载所有迁移"""
        
        migrations = []
        
        # 这里可以动态加载迁移文件或使用预定义的迁移
        migrations.extend([
            WorkflowTableMigration("1.1.0", "Add workflow execution tracking"),
            WebSocketConfigMigration("1.1.1", "Add WebSocket configuration"),
            ErrorLogMigration("1.2.0", "Add error logging table"),
            UserPreferencesMigration("1.2.1", "Add user preferences table")
        ])
        
        return sorted(migrations, key=lambda m: m.version)
    
    def migrate(self, target_version: str = None) -> bool:
        """执行迁移"""
        
        pending = self.get_pending_migrations()
        
        if target_version:
            pending = [m for m in pending if m.version <= target_version]
        
        if not pending:
            self.logger.info("No pending migrations")
            return True
        
        try:
            with sqlite3.connect(self.db_path) as conn:
                for migration in pending:
                    self.logger.info(f"Applying migration {migration.version}: {migration.description}")
                    
                    # 执行迁移
                    migration.up(conn)
                    
                    # 记录迁移
                    conn.execute(
                        "INSERT INTO schema_migrations (version, description) VALUES (?, ?)",
                        (migration.version, migration.description)
                    )
                    
                    conn.commit()
                    self.logger.info(f"Migration {migration.version} applied successfully")
            
            return True
            
        except Exception as e:
            self.logger.error(f"Migration failed: {e}")
            return False
    
    def rollback(self, target_version: str) -> bool:
        """回滚到指定版本"""
        
        applied = self.get_applied_migrations()
        to_rollback = [v for v in applied if v > target_version]
        
        if not to_rollback:
            self.logger.info(f"Already at or below version {target_version}")
            return True
        
        try:
            with sqlite3.connect(self.db_path) as conn:
                for version in reversed(to_rollback):
                    migration = self._get_migration_by_version(version)
                    if migration:
                        self.logger.info(f"Rolling back migration {version}")
                        
                        migration.down(conn)
                        
                        conn.execute(
                            "DELETE FROM schema_migrations WHERE version = ?",
                            (version,)
                        )
                        
                        conn.commit()
                        self.logger.info(f"Migration {version} rolled back successfully")
            
            return True
            
        except Exception as e:
            self.logger.error(f"Rollback failed: {e}")
            return False

# backend/open_webui/migrations/config_migrator.py
import json
import yaml
import os
from typing import Dict, Any, Optional
import logging

class ConfigMigrator:
    """配置文件迁移器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.migration_rules = self._load_migration_rules()
    
    def _load_migration_rules(self) -> Dict[str, Any]:
        """加载配置迁移规则"""
        
        return {
            "1.0.0_to_1.1.0": {
                "add": {
                    "websocket": {
                        "enabled": False,
                        "host": "localhost",
                        "port": 8765,
                        "ping_interval": 30
                    }
                },
                "rename": {
                    "n8n_webhook_url": "n8n.webhook_base_url"
                },
                "remove": [
                    "deprecated_setting"
                ]
            },
            "1.1.0_to_1.2.0": {
                "add": {
                    "llm_response_fixer": {
                        "enabled": False,
                        "max_retries": 3,
                        "timeout": 30
                    },
                    "error_handling": {
                        "circuit_breaker": {
                            "failure_threshold": 5,
                            "recovery_timeout": 60
                        }
                    }
                },
                "modify": {
                    "n8n.timeout": 300  # 增加超时时间
                }
            }
        }
    
    def migrate_config(self, config_path: str, from_version: str, to_version: str) -> bool:
        """迁移配置文件"""
        
        if not os.path.exists(config_path):
            self.logger.error(f"Config file not found: {config_path}")
            return False
        
        try:
            # 备份原配置
            backup_path = f"{config_path}.backup.{from_version}"
            self._backup_config(config_path, backup_path)
            
            # 加载当前配置
            config = self._load_config(config_path)
            
            # 应用迁移规则
            migration_key = f"{from_version}_to_{to_version}"
            if migration_key in self.migration_rules:
                config = self._apply_migration_rules(config, self.migration_rules[migration_key])
            
            # 保存新配置
            self._save_config(config, config_path)
            
            self.logger.info(f"Config migrated from {from_version} to {to_version}")
            return True
            
        except Exception as e:
            self.logger.error(f"Config migration failed: {e}")
            # 恢复备份
            if os.path.exists(backup_path):
                os.rename(backup_path, config_path)
            return False
    
    def _backup_config(self, source_path: str, backup_path: str):
        """备份配置文件"""
        
        import shutil
        shutil.copy2(source_path, backup_path)
        self.logger.info(f"Config backed up to {backup_path}")
    
    def _load_config(self, config_path: str) -> Dict[str, Any]:
        """加载配置文件"""
        
        with open(config_path, 'r', encoding='utf-8') as f:
            if config_path.endswith('.json'):
                return json.load(f)
            elif config_path.endswith(('.yml', '.yaml')):
                return yaml.safe_load(f)
            else:
                raise ValueError(f"Unsupported config format: {config_path}")
    
    def _save_config(self, config: Dict[str, Any], config_path: str):
        """保存配置文件"""
        
        with open(config_path, 'w', encoding='utf-8') as f:
            if config_path.endswith('.json'):
                json.dump(config, f, indent=2, ensure_ascii=False)
            elif config_path.endswith(('.yml', '.yaml')):
                yaml.dump(config, f, default_flow_style=False, allow_unicode=True)
    
    def _apply_migration_rules(self, config: Dict[str, Any], rules: Dict[str, Any]) -> Dict[str, Any]:
        """应用迁移规则"""
        
        # 添加新配置
        if "add" in rules:
            config = self._deep_merge(config, rules["add"])
        
        # 重命名配置
        if "rename" in rules:
            for old_key, new_key in rules["rename"].items():
                if old_key in config:
                    self._set_nested_value(config, new_key, config[old_key])
                    del config[old_key]
        
        # 修改配置
        if "modify" in rules:
            for key, value in rules["modify"].items():
                self._set_nested_value(config, key, value)
        
        # 删除配置
        if "remove" in rules:
            for key in rules["remove"]:
                if key in config:
                    del config[key]
        
        return config
    
    def _deep_merge(self, dict1: Dict[str, Any], dict2: Dict[str, Any]) -> Dict[str, Any]:
        """深度合并字典"""
        
        result = dict1.copy()
        
        for key, value in dict2.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = self._deep_merge(result[key], value)
            else:
                result[key] = value
        
        return result
    
    def _set_nested_value(self, config: Dict[str, Any], key_path: str, value: Any):
        """设置嵌套配置值"""
        
        keys = key_path.split('.')
        current = config
        
        for key in keys[:-1]:
            if key not in current:
                current[key] = {}
            current = current[key]
        
        current[keys[-1]] = value

import json
from typing import Dict, Any, List, Optional, Tuple
import logging

class CompatibilityChecker:
    """兼容性检查器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.requirements = self._load_requirements()
    
    def _load_requirements(self) -> Dict[str, Dict[str, str]]:
        """加载兼容性要求"""
        
        return {
            "1.0.0": {
                "python": ">=3.8",
                "n8n": ">=1.0.0",
                "openwebui": ">=0.1.0"
            },
            "1.1.0": {
                "python": ">=3.8", 
                "n8n": ">=1.0.0",
                "openwebui": ">=0.1.0",
                "websockets": ">=10.0"
            },
            "1.2.0": {
                "python": ">=3.9",
                "n8n": ">=1.1.0", 
                "openwebui": ">=0.2.0",
                "websockets": ">=10.0",
                "transformers": ">=4.20.0"
            }
        }
    
    def check_compatibility(self, target_version: str) -> Tuple[bool, List[str]]:
        """检查兼容性"""
        
        if target_version not in self.requirements:
            return False, [f"Unknown version: {target_version}"]
        
        requirements = self.requirements[target_version]
        issues = []
        
        # 检查Python版本
        python_issue = self._check_python_version(requirements.get("python"))
        if python_issue:
            issues.append(python_issue)
        
        # 检查n8n版本
        n8n_issue = self._check_n8n_version(requirements.get("n8n"))
        if n8n_issue:
            issues.append(n8n_issue)
        
        # 检查其他依赖
        for package, version_req in requirements.items():
            if package not in ["python", "n8n", "openwebui"]:
                issue = self._check_package_version(package, version_req)
                if issue:
                    issues.append(issue)
        
        return len(issues) == 0, issues
    
    def _check_python_version(self, requirement: str) -> Optional[str]:
        """检查Python版本"""
        
        if not requirement:
            return None
        
        import sys
        current_version = f"{sys.version_info.major}.{sys.version_info.minor}.{sys.version_info.micro}"
        
        if not self._version_satisfies(current_version, requirement):
            return f"Python version {current_version} does not satisfy {requirement}"
        
        return None
    
    def _check_n8n_version(self, requirement: str) -> Optional[str]:
        """检查n8n版本"""
        
        if not requirement:
            return None
        
        try:
            # 通过API检查n8n版本
            import requests
            response = requests.get("http://localhost:5678/rest/active-workflows", timeout=5)
            
            if response.status_code == 200:
                # n8n可访问，假设版本兼容
                return None
            else:
                return "Cannot connect to n8n instance"
                
        except Exception as e:
            return f"n8n connectivity check failed: {str(e)}"
    
    def _check_package_version(self, package: str, requirement: str) -> Optional[str]:
        """检查Python包版本"""
        
        try:
            import pkg_resources
            installed_version = pkg_resources.get_distribution(package).version
            
            if not self._version_satisfies(installed_version, requirement):
                return f"Package {package} version {installed_version} does not satisfy {requirement}"
            
            return None
            
        except pkg_resources.DistributionNotFound:
            return f"Package {package} is not installed (required: {requirement})"
        except Exception as e:
            return f"Failed to check {package} version: {str(e)}"
    
    def _version_satisfies(self, version: str, requirement: str) -> bool:
        """检查版本是否满足要求"""
        
        try:
            import packaging.specifiers
            import packaging.version
            
            spec = packaging.specifiers.SpecifierSet(requirement)
            return packaging.version.Version(version) in spec
            
        except Exception:
            # 简单的版本比较作为后备
            return version >= requirement.replace(">=", "").replace(">", "").strip()

# 全局兼容性检查器
compatibility_checker = CompatibilityChecker()

# backend/open_webui/services/upgrade_coordinator.py
from typing import Dict, Any, List, Optional, Callable
from datetime import datetime
import asyncio
import logging

class UpgradeCoordinator:
    """升级协调器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.upgrade_steps = []
        self.rollback_steps = []
        self.status_callbacks = []
    
    async def coordinate_upgrade(self, from_version: str, to_version: str, 
                               options: Dict[str, Any] = None) -> bool:
        """协调升级过程"""
        
        options = options or {}
        
        try:
            # 1. 预检查
            self._notify_status("pre_check", "开始升级前检查")
            
            if not await self._pre_upgrade_check(from_version, to_version):
                raise Exception("Pre-upgrade check failed")
            
            # 2. 准备升级
            self._notify_status("preparing", "准备升级环境")
            await self._prepare_upgrade(from_version, to_version, options)
            
            # 3. 执行升级步骤
            self._notify_status("upgrading", "执行升级")
            await self._execute_upgrade_steps(from_version, to_version)
            
            # 4. 验证升级
            self._notify_status("validating", "验证升级结果")
            if not await self._validate_upgrade(to_version):
                raise Exception("Upgrade validation failed")
            
            # 5. 完成升级
            self._notify_status("completed", "升级完成")
            await self._finalize_upgrade(to_version)
            
            return True
            
        except Exception as e:
            self.logger.error(f"Upgrade failed: {e}")
            
            # 执行回滚
            self._notify_status("rolling_back", f"升级失败，正在回滚: {str(e)}")
            await self._rollback_upgrade(from_version)
            
            return False
    
    async def _pre_upgrade_check(self, from_version: str, to_version: str) -> bool:
        """升级前检查"""
        
        # 兼容性检查
        compatible, issues = compatibility_checker.check_compatibility(to_version)
        if not compatible:
            self.logger.error(f"Compatibility issues: {issues}")
            return False
        
        # 磁盘空间检查
        if not self._check_disk_space():
            self.logger.error("Insufficient disk space")
            return False
        
        # 备份检查
        if not await self._create_backup():
            self.logger.error("Failed to create backup")
            return False
        
        return True
    
    async def _prepare_upgrade(self, from_version: str, to_version: str, options: Dict[str, Any]):
        """准备升级"""
        
        # 停止相关服务
        await self._stop_services()
        
        # 准备迁移脚本
        self._prepare_migration_scripts(from_version, to_version)
        
        # 设置功能开关
        self._prepare_feature_flags(to_version, options)
    
    async def _execute_upgrade_steps(self, from_version: str, to_version: str):
        """执行升级步骤"""
        
        steps = [
            ("database_migration", self._migrate_database),
            ("config_migration", self._migrate_config),
            ("code_deployment", self._deploy_code),
            ("service_restart", self._restart_services)
        ]
        
        for step_name, step_func in steps:
            self._notify_status("upgrading", f"执行步骤: {step_name}")
            
            try:
                await step_func(from_version, to_version)
                self.rollback_steps.append((step_name, step_func))
                
            except Exception as e:
                self.logger.error(f"Upgrade step {step_name} failed: {e}")
                raise
    
    async def _validate_upgrade(self, to_version: str) -> bool:
        """验证升级结果"""
        
        # 检查服务状态
        if not await self._check_service_health():
            return False
        
        # 检查数据完整性
        if not await self._check_data_integrity():
            return False
        
        # 功能测试
        if not await self._run_smoke_tests():
            return False
        
        return True
    
    def _notify_status(self, status: str, message: str):
        """通知升级状态"""
        
        status_info = {
            "status": status,
            "message": message,
            "timestamp": datetime.now().isoformat()
        }
        
        for callback in self.status_callbacks:
            try:
                callback(status_info)
            except Exception as e:
                self.logger.error(f"Status callback failed: {e}")
    
    def add_status_callback(self, callback: Callable):
        """添加状态回调"""
        self.status_callbacks.append(callback)

# 全局升级协调器
