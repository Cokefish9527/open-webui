## 1. 问题分析

### 1.1 现有架构问题

当前系统采用HTTP WebHook方式与n8n工作流通信，存在以下限制：

1. **单向通信**：HTTP请求-响应模式无法实现双向实时通信
2. **状态盲区**：无法获取工作流执行过程中的实时状态
3. **超时风险**：长时间运行的工作流可能导致HTTP超时
4. **用户体验差**：用户无法感知工作流执行进度

### 1.2 优化目标

- **阶段一**：优化HTTP通信，增加结构化处理和错误恢复
- **阶段二**：引入WebSocket双向通信，实现实时状态反馈

## 2. 阶段一：HTTP协议优化

### 2.1 结构化响应处理器

``python
# backend/open_webui/utils/response_processor.py
import json
import re
from typing import Dict, Any, Optional, Union
from datetime import datetime
import logging

class ResponseProcessor:
    """n8n响应结构化处理器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        
    def process_n8n_response(self, raw_response: Union[str, Dict[str, Any]]) -> Dict[str, Any]:
        """处理n8n原始响应，返回结构化数据"""
        
        try:
            # 1. 数据类型标准化
            if isinstance(raw_response, str):
                # 尝试解析JSON字符串
                try:
                    parsed_response = json.loads(raw_response)
                except json.JSONDecodeError:
                    # 如果不是JSON，作为纯文本处理
                    parsed_response = {"content": raw_response, "type": "text"}
            else:
                parsed_response = raw_response
            
            # 2. 响应结构验证和修复
            structured_response = self._validate_and_fix_structure(parsed_response)
            
            # 3. 内容格式化
            formatted_response = self._format_content(structured_response)
            
            # 4. 添加元数据
            final_response = self._add_metadata(formatted_response)
            
            return final_response
            
        except Exception as e:
            self.logger.error(f"Response processing failed: {e}")
            return self._create_error_response(str(e), raw_response)
    
    def _validate_and_fix_structure(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """验证并修复响应结构"""
        
        # 标准响应结构
        standard_structure = {
            "success": True,
            "data": {},
            "message": "",
            "timestamp": datetime.now().isoformat(),
            "workflow_info": {}
        }
        
        # 检查必要字段
        if not isinstance(response, dict):
            return {
                **standard_structure,
                "data": {"content": str(response)},
                "message": "Response converted from non-dict type"
            }
        
        # 修复常见的结构问题
        fixed_response = standard_structure.copy()
        
        # 处理成功标志
        if "success" in response:
            fixed_response["success"] = bool(response["success"])
        elif "error" in response or "Error" in response:
            fixed_response["success"] = False
        
        # 处理数据字段
        if "data" in response:
            fixed_response["data"] = response["data"]
        elif "result" in response:
            fixed_response["data"] = response["result"]
        elif "output" in response:
            fixed_response["data"] = response["output"]
        else:
            # 将整个响应作为数据
            fixed_response["data"] = {k: v for k, v in response.items() 
                                    if k not in ["success", "message", "timestamp"]}
        
        # 处理消息字段
        if "message" in response:
            fixed_response["message"] = str(response["message"])
        elif "error" in response:
            fixed_response["message"] = str(response["error"])
            fixed_response["success"] = False
        
        # 处理工作流信息
        workflow_fields = ["workflow_id", "workflow_type", "execution_id", "node_info"]
        workflow_info = {}
        for field in workflow_fields:
            if field in response:
                workflow_info[field] = response[field]
        
        if workflow_info:
            fixed_response["workflow_info"] = workflow_info
        
        return fixed_response
    
    def _format_content(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """格式化响应内容"""
        
        data = response.get("data", {})
        
        # 处理不同类型的内容
        if isinstance(data, dict):
            # 格式化文本内容
            if "content" in data and isinstance(data["content"], str):
                data["content"] = self._format_text_content(data["content"])
            
            # 格式化列表数据
            if "items" in data and isinstance(data["items"], list):
                data["items"] = self._format_list_items(data["items"])
            
            # 格式化关键词数据
            if "keywords" in data:
                data["keywords"] = self._format_keywords(data["keywords"])
        
        response["data"] = data
        return response
    
    def _format_text_content(self, content: str) -> str:
        """格式化文本内容"""
        
        # 清理多余的空白字符
        content = re.sub(r'\s+', ' ', content.strip())
        
        # 修复常见的格式问题
        content = re.sub(r'([.!?])\s*([A-Z])', r'\1 \2', content)  # 句子间距
        content = re.sub(r'([,;:])\s*', r'\1 ', content)  # 标点后空格
        
        return content
    
    def _format_list_items(self, items: list) -> list:
        """格式化列表项"""
        
        formatted_items = []
        for item in items:
            if isinstance(item, str):
                formatted_items.append(self._format_text_content(item))
            elif isinstance(item, dict):
                # 递归处理字典项
                formatted_item = {}
                for key, value in item.items():
                    if isinstance(value, str):
                        formatted_item[key] = self._format_text_content(value)
                    else:
                        formatted_item[key] = value
                formatted_items.append(formatted_item)
            else:
                formatted_items.append(item)
        
        return formatted_items
    
    def _format_keywords(self, keywords: Union[str, list, dict]) -> list:
        """格式化关键词数据"""
        
        if isinstance(keywords, str):
            # 从字符串中提取关键词
            keyword_list = re.split(r'[,;，；\n]', keywords)
            return [kw.strip() for kw in keyword_list if kw.strip()]
        
        elif isinstance(keywords, list):
            return [str(kw).strip() for kw in keywords if str(kw).strip()]
        
        elif isinstance(keywords, dict):
            # 如果是字典，提取值
            all_keywords = []
            for value in keywords.values():
                if isinstance(value, (list, str)):
                    all_keywords.extend(self._format_keywords(value))
            return all_keywords
        
        return []
    
    def _add_metadata(self, response: Dict[str, Any]) -> Dict[str, Any]:
        """添加元数据"""
        
        response["metadata"] = {
            "processed_at": datetime.now().isoformat(),
            "processor_version": "1.0",
            "content_type": self._detect_content_type(response["data"]),
            "data_size": len(str(response["data"]))
        }
        
        return response
    
    def _detect_content_type(self, data: Dict[str, Any]) -> str:
        """检测内容类型"""
        
        if "keywords" in data:
            return "keywords"
        elif "content" in data:
            return "text_content"
        elif "items" in data:
            return "list_data"
        elif "analysis" in data:
            return "analysis_result"
        else:
            return "generic"
    
    def _create_error_response(self, error_message: str, raw_data: Any) -> Dict[str, Any]:
        """创建错误响应"""
        
        return {
            "success": False,
            "data": {},
            "message": f"Response processing error: {error_message}",
            "timestamp": datetime.now().isoformat(),
            "workflow_info": {},
            "metadata": {
                "processed_at": datetime.now().isoformat(),
                "processor_version": "1.0",
                "content_type": "error",
                "raw_data": str(raw_data)[:1000]  # 限制原始数据长度
            }
        }

### 2.2 HTTP通信增强器

``python
# backend/open_webui/utils/http_enhancer.py
import aiohttp
import asyncio
import json
from typing import Dict, Any, Optional, Callable
from datetime import datetime, timedelta
import logging
from .response_processor import response_processor

class HTTPEnhancer:
    """HTTP通信增强器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.session_pool = {}
        self.retry_config = {
            "max_retries": 3,
            "base_delay": 1.0,
            "max_delay": 30.0,
            "backoff_factor": 2.0
        }
        self.timeout_config = {
            "connect_timeout": 10.0,
            "read_timeout": 300.0,  # 5分钟
            "total_timeout": 600.0   # 10分钟
        }
    
    async def enhanced_request(self, 
                             url: str,
                             payload: Dict[str, Any],
                             session_id: str = None,
                             progress_callback: Callable = None) -> Dict[str, Any]:
        """增强的HTTP请求"""
        
        request_id = f"{session_id or 'anonymous'}_{datetime.now().timestamp()}"
        
        try:
            # 1. 预处理请求
            processed_payload = await self._preprocess_request(payload, request_id)
            
            # 2. 执行请求（带重试）
            raw_response = await self._execute_with_retry(
                url, processed_payload, request_id, progress_callback
            )
            
            # 3. 后处理响应
            processed_response = await self._postprocess_response(
                raw_response, request_id
            )
            
            return processed_response
            
        except Exception as e:
            self.logger.error(f"Enhanced request failed for {request_id}: {e}")
            return self._create_error_response(str(e), request_id)
    
    async def _preprocess_request(self, payload: Dict[str, Any], request_id: str) -> Dict[str, Any]:
        """预处理请求数据"""
        
        processed_payload = payload.copy()
        
        # 添加请求元数据
        processed_payload["_metadata"] = {
            "request_id": request_id,
            "timestamp": datetime.now().isoformat(),
            "client_version": "openwebui-enhanced-1.0"
        }
        
        # 数据验证和清理
        processed_payload = self._validate_and_clean_payload(processed_payload)
        
        return processed_payload
    
    def _validate_and_clean_payload(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """验证和清理请求载荷"""
        
        # 移除空值
        cleaned_payload = {}
        for key, value in payload.items():
            if value is not None and value != "":
                if isinstance(value, str):
                    cleaned_payload[key] = value.strip()
                else:
                    cleaned_payload[key] = value
        
        # 确保必要字段存在
        if "input" not in cleaned_payload:
            cleaned_payload["input"] = ""
        
        return cleaned_payload
    
    async def _execute_with_retry(self, 
                                url: str,
                                payload: Dict[str, Any],
                                request_id: str,
                                progress_callback: Callable = None) -> Dict[str, Any]:
        """带重试的请求执行"""
        
        last_exception = None
        
        for attempt in range(self.retry_config["max_retries"]):
            try:
                if progress_callback:
                    await progress_callback(f"尝试第 {attempt + 1} 次请求...")
                
                # 创建会话
                timeout = aiohttp.ClientTimeout(
                    connect=self.timeout_config["connect_timeout"],
                    sock_read=self.timeout_config["read_timeout"],
                    total=self.timeout_config["total_timeout"]
                )
                
                async with aiohttp.ClientSession(timeout=timeout) as session:
                    async with session.post(url, json=payload) as response:
                        
                        if response.status == 200:
                            response_data = await response.json()
                            
                            if progress_callback:
                                await progress_callback("请求成功完成")
                            
                            return response_data
                        
                        else:
                            error_text = await response.text()
                            raise aiohttp.ClientResponseError(
                                request_info=response.request_info,
                                history=response.history,
                                status=response.status,
                                message=f"HTTP {response.status}: {error_text}"
                            )
            
            except Exception as e:
                last_exception = e
                self.logger.warning(f"Request attempt {attempt + 1} failed: {e}")
                
                if attempt < self.retry_config["max_retries"] - 1:
                    # 计算延迟时间
                    delay = min(
                        self.retry_config["base_delay"] * (
                            self.retry_config["backoff_factor"] ** attempt
                        ),
                        self.retry_config["max_delay"]
                    )
                    
                    if progress_callback:
                        await progress_callback(f"请求失败，{delay}秒后重试...")
                    
                    await asyncio.sleep(delay)
        
        # 所有重试都失败
        raise last_exception
    
    async def _postprocess_response(self, 
                                  raw_response: Dict[str, Any],
                                  request_id: str) -> Dict[str, Any]:
        """后处理响应数据"""
        
        # 使用响应处理器进行结构化处理
        processed_response = response_processor.process_n8n_response(raw_response)
        
        # 添加请求追踪信息
        processed_response["request_info"] = {
            "request_id": request_id,
            "processed_at": datetime.now().isoformat()
        }
        
        return processed_response
    
    def _create_error_response(self, error_message: str, request_id: str) -> Dict[str, Any]:
        """创建错误响应"""
        
        return {
            "success": False,
            "data": {},
            "message": f"HTTP request failed: {error_message}",
            "timestamp": datetime.now().isoformat(),
            "request_info": {
                "request_id": request_id,
                "processed_at": datetime.now().isoformat()
            },
            "metadata": {
                "content_type": "error",
                "processor_version": "1.0"
            }
        }

# backend/open_webui/websocket/client_manager.py
import asyncio
import json
import logging
from typing import Dict, Any, Set, Optional, Callable
from datetime import datetime
import websockets
from websockets.exceptions import ConnectionClosed, WebSocketException

class WebSocketClientManager:
    """WebSocket客户端管理器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.connections: Dict[str, websockets.WebSocketServerProtocol] = {}
        self.user_sessions: Dict[str, Set[str]] = {}  # user_id -> session_ids
        self.session_users: Dict[str, str] = {}      # session_id -> user_id
        self.message_handlers: Dict[str, Callable] = {}
        
    async def register_connection(self, websocket: websockets.WebSocketServerProtocol, 
                                user_id: str, session_id: str):
        """注册WebSocket连接"""
        
        self.connections[session_id] = websocket
        self.session_users[session_id] = user_id
        
        if user_id not in self.user_sessions:
            self.user_sessions[user_id] = set()
        self.user_sessions[user_id].add(session_id)
        
        self.logger.info(f"WebSocket connection registered: user={user_id}, session={session_id}")
        
        # 发送连接确认
        await self.send_to_session(session_id, {
            "type": "connection_established",
            "session_id": session_id,
            "timestamp": datetime.now().isoformat()
        })
    
    async def unregister_connection(self, session_id: str):
        """注销WebSocket连接"""
        
        if session_id in self.connections:
            del self.connections[session_id]
        
        if session_id in self.session_users:
            user_id = self.session_users[session_id]
            del self.session_users[session_id]
            
            if user_id in self.user_sessions:
                self.user_sessions[user_id].discard(session_id)
                if not self.user_sessions[user_id]:
                    del self.user_sessions[user_id]
        
        self.logger.info(f"WebSocket connection unregistered: session={session_id}")
    
    async def send_to_session(self, session_id: str, message: Dict[str, Any]) -> bool:
        """向特定会话发送消息"""
        
        if session_id not in self.connections:
            return False
        
        try:
            websocket = self.connections[session_id]
            await websocket.send(json.dumps(message))
            return True
        
        except (ConnectionClosed, WebSocketException) as e:
            self.logger.warning(f"Failed to send message to session {session_id}: {e}")
            await self.unregister_connection(session_id)
            return False
    
    async def send_to_user(self, user_id: str, message: Dict[str, Any]) -> int:
        """向用户的所有会话发送消息"""
        
        if user_id not in self.user_sessions:
            return 0
        
        sent_count = 0
        sessions_to_remove = []
        
        for session_id in self.user_sessions[user_id].copy():
            if await self.send_to_session(session_id, message):
                sent_count += 1
            else:
                sessions_to_remove.append(session_id)
        
        # 清理失效的会话
        for session_id in sessions_to_remove:
            await self.unregister_connection(session_id)
        
        return sent_count
    
    async def broadcast_to_all(self, message: Dict[str, Any]) -> int:
        """向所有连接广播消息"""
        
        sent_count = 0
        sessions_to_remove = []
        
        for session_id in list(self.connections.keys()):
            if await self.send_to_session(session_id, message):
                sent_count += 1
            else:
                sessions_to_remove.append(session_id)
        
        # 清理失效的会话
        for session_id in sessions_to_remove:
            await self.unregister_connection(session_id)
        
        return sent_count
    
    def register_message_handler(self, message_type: str, handler: Callable):
        """注册消息处理器"""
        self.message_handlers[message_type] = handler
    
    async def handle_message(self, session_id: str, message: Dict[str, Any]):
        """处理收到的消息"""
        
        message_type = message.get("type")
        if message_type in self.message_handlers:
            try:
                await self.message_handlers[message_type](session_id, message)
            except Exception as e:
                self.logger.error(f"Message handler error for type {message_type}: {e}")
                
                # 发送错误响应
                await self.send_to_session(session_id, {
                    "type": "error",
                    "message": f"Handler error: {str(e)}",
                    "original_message": message,
                    "timestamp": datetime.now().isoformat()
                })
        else:
            self.logger.warning(f"No handler for message type: {message_type}")
    
    def get_connection_stats(self) -> Dict[str, Any]:
        """获取连接统计信息"""
        
        return {
            "total_connections": len(self.connections),
            "total_users": len(self.user_sessions),
            "connections_per_user": {
                user_id: len(sessions) 
                for user_id, sessions in self.user_sessions.items()
            },
            "timestamp": datetime.now().isoformat()
        }

# backend/open_webui/websocket/status_broadcaster.py
import asyncio
import json
from typing import Dict, Any, Optional
from datetime import datetime
from .client_manager import ws_client_manager

class StatusBroadcaster:
    """实时状态广播器"""
    
    def __init__(self):
        self.workflow_subscribers: Dict[str, set] = {}  # workflow_id -> session_ids
        
        # 注册消息处理器
        ws_client_manager.register_message_handler("subscribe_workflow", self._handle_subscribe)
        ws_client_manager.register_message_handler("unsubscribe_workflow", self._handle_unsubscribe)
    
    async def _handle_subscribe(self, session_id: str, message: Dict[str, Any]):
        """处理工作流订阅"""
        
        workflow_id = message.get("workflow_id")
        if not workflow_id:
            return
        
        if workflow_id not in self.workflow_subscribers:
            self.workflow_subscribers[workflow_id] = set()
        
        self.workflow_subscribers[workflow_id].add(session_id)
        
        # 发送订阅确认
        await ws_client_manager.send_to_session(session_id, {
            "type": "subscription_confirmed",
            "workflow_id": workflow_id,
            "timestamp": datetime.now().isoformat()
        })
    
    async def _handle_unsubscribe(self, session_id: str, message: Dict[str, Any]):
        """处理工作流取消订阅"""
        
        workflow_id = message.get("workflow_id")
        if not workflow_id or workflow_id not in self.workflow_subscribers:
            return
        
        self.workflow_subscribers[workflow_id].discard(session_id)
        
        # 如果没有订阅者了，清理
        if not self.workflow_subscribers[workflow_id]:
            del self.workflow_subscribers[workflow_id]
        
        # 发送取消订阅确认
        await ws_client_manager.send_to_session(session_id, {
            "type": "unsubscription_confirmed",
            "workflow_id": workflow_id,
            "timestamp": datetime.now().isoformat()
        })
    
    async def broadcast_workflow_status(self, workflow_id: str, status_data: Dict[str, Any]):
        """广播工作流状态更新"""
        
        if workflow_id not in self.workflow_subscribers:
            return
        
        message = {
            "type": "workflow_status_update",
            "workflow_id": workflow_id,
            "status": status_data,
            "timestamp": datetime.now().isoformat()
        }
        
        # 向所有订阅者发送状态更新
        subscribers = self.workflow_subscribers[workflow_id].copy()
        for session_id in subscribers:
            success = await ws_client_manager.send_to_session(session_id, message)
            if not success:
                # 移除失效的订阅者
                self.workflow_subscribers[workflow_id].discard(session_id)
    
    async def broadcast_workflow_progress(self, workflow_id: str, progress_data: Dict[str, Any]):
        """广播工作流进度更新"""
        
        if workflow_id not in self.workflow_subscribers:
            return
        
        message = {
            "type": "workflow_progress_update",
            "workflow_id": workflow_id,
            "progress": progress_data,
            "timestamp": datetime.now().isoformat()
        }
        
        # 向所有订阅者发送进度更新
        subscribers = self.workflow_subscribers[workflow_id].copy()
        for session_id in subscribers:
            success = await ws_client_manager.send_to_session(session_id, message)
            if not success:
                # 移除失效的订阅者
                self.workflow_subscribers[workflow_id].discard(session_id)
    
    async def broadcast_workflow_error(self, workflow_id: str, error_data: Dict[str, Any]):
        """广播工作流错误"""
        
        if workflow_id not in self.workflow_subscribers:
            return
        
        message = {
            "type": "workflow_error",
            "workflow_id": workflow_id,
            "error": error_data,
            "timestamp": datetime.now().isoformat()
        }
        
        # 向所有订阅者发送错误信息

        user_id = None

                await websocket.close(code=4001, reason="Authentication required")

            session_id = auth_data.get("session_id")

                await websocket.close(code=4002, reason="Invalid authentication data")

                try:

// src/lib/websocket/WorkflowSocketClient.js
class WorkflowSocketClient {
    constructor() {
        this.socket = null;
        this.isConnected = false;
        this.reconnectAttempts = 0;
        this.maxReconnectAttempts = 5;
        this.reconnectDelay = 1000;
        this.messageHandlers = new Map();
        this.subscriptions = new Set();
        
        // 绑定事件处理器
        this.onOpen = this.onOpen.bind(this);
        this.onMessage = this.onMessage.bind(this);
        this.onClose = this.onClose.bind(this);
        this.onError = this.onError.bind(this);
    }
    
    async connect(userId, sessionId) {
        if (this.isConnected) {
            return;
        }
        
        try {
            const wsUrl = `ws://localhost:8765`;
            this.socket = new WebSocket(wsUrl);
            
            this.socket.addEventListener('open', this.onOpen);
            this.socket.addEventListener('message', this.onMessage);
            this.socket.addEventListener('close', this.onClose);
            this.socket.addEventListener('error', this.onError);
            
            // 等待连接建立
            await new Promise((resolve, reject) => {
                const timeout = setTimeout(() => {
                    reject(new Error('Connection timeout'));
                }, 10000);
                
                this.socket.addEventListener('open', () => {
                    clearTimeout(timeout);
                    resolve();
                }, { once: true });
                
                this.socket.addEventListener('error', (error) => {
                    clearTimeout(timeout);
                    reject(error);
                }, { once: true });
            });
            
            // 发送认证消息
            await this.authenticate(userId, sessionId);
            
        } catch (error) {
            console.error('WebSocket connection failed:', error);
            throw error;
        }
    }
    
    async authenticate(userId, sessionId) {
        const authMessage = {
            type: 'authenticate',
            user_id: userId,
            session_id: sessionId,
            timestamp: new Date().toISOString()
        };
        
        this.send(authMessage);
        
        // 等待认证确认
        return new Promise((resolve, reject) => {
            const timeout = setTimeout(() => {
                reject(new Error('Authentication timeout'));
            }, 5000);
            
            const handler = (data) => {
                if (data.type === 'connection_established') {
                    clearTimeout(timeout);
                    this.off('message', handler);
                    resolve();
                }
            };
            
            this.on('message', handler);
        });
    }
    
    disconnect() {
        if (this.socket) {
            this.socket.removeEventListener('open', this.onOpen);
            this.socket.removeEventListener('message', this.onMessage);
            this.socket.removeEventListener('close', this.onClose);
            this.socket.removeEventListener('error', this.onError);
            
            this.socket.close();
            this.socket = null;
        }
        
        this.isConnected = false;
        this.subscriptions.clear();
    }
    
    send(message) {
        if (this.isConnected && this.socket) {
            this.socket.send(JSON.stringify(message));
        } else {
            console.warn('WebSocket not connected, message not sent:', message);
        }
    }
    
    subscribeToWorkflow(workflowId) {
        const message = {
            type: 'subscribe_workflow',
            workflow_id: workflowId,
            timestamp: new Date().toISOString()
        };
        
        this.send(message);
        this.subscriptions.add(workflowId);
    }
    
    unsubscribeFromWorkflow(workflowId) {
        const message = {
            type: 'unsubscribe_workflow',
            workflow_id: workflowId,
            timestamp: new Date().toISOString()
        };
        
        this.send(message);
        this.subscriptions.delete(workflowId);
    }
    
    on(event, handler) {
        if (!this.messageHandlers.has(event)) {
            this.messageHandlers.set(event, []);
        }
        this.messageHandlers.get(event).push(handler);
    }
    
    off(event, handler) {
        if (this.messageHandlers.has(event)) {
            const handlers = this.messageHandlers.get(event);
            const index = handlers.indexOf(handler);
            if (index > -1) {
                handlers.splice(index, 1);
            }
        }
    }
    
    emit(event, data) {
        if (this.messageHandlers.has(event)) {
            this.messageHandlers.get(event).forEach(handler => {
                try {
                    handler(data);
                } catch (error) {
                    console.error('Error in message handler:', error);
                }
            });
        }
    }
    
    onOpen(event) {
        console.log('WebSocket connected');
        this.isConnected = true;
        this.reconnectAttempts = 0;
        this.emit('connected', event);
    }
    
    onMessage(event) {
        try {
            const data = JSON.parse(event.data);
            console.log('WebSocket message received:', data);
            
            // 触发通用消息事件
            this.emit('message', data);
            
            // 触发特定类型的事件
            if (data.type) {
                this.emit(data.type, data);
            }
            
        } catch (error) {
            console.error('Error parsing WebSocket message:', error);
        }
    }
    
    onClose(event) {
        console.log('WebSocket disconnected:', event.code, event.reason);
        this.isConnected = false;
        this.emit('disconnected', event);
        
        // 自动重连
        if (this.reconnectAttempts < this.maxReconnectAttempts) {
            this.reconnectAttempts++;
            const delay = this.reconnectDelay * Math.pow(2, this.reconnectAttempts - 1);
            
            console.log(`Attempting to reconnect in ${delay}ms (attempt ${this.reconnectAttempts})`);
            
            setTimeout(() => {
                this.connect();
            }, delay);
        }
    }
    
    onError(event) {
        console.error('WebSocket error:', event);
        this.emit('error', event);
    }
}

from typing import Dict, Any, Optional, Callable
from .workflow_state_manager import WorkflowStateManager, WorkflowStatus
from ..utils.http_enhancer import http_enhancer

class HybridCommunicationService:

        self.state_manager = WorkflowStateManager()
        
        # 注册状态变更回调
        self.state_manager.subscribe_to_status_changes("*", self._on_status_change)
    
    async def execute_workflow_with_realtime_feedback(self,
                                                    url: str,
                                                    payload: Dict[str, Any],
                                                    user_id: str,
                                                    workflow_type: str) -> Dict[str, Any]:
        """执行工作流并提供实时反馈"""
        
        # 1. 创建工作流会话
        session_id = await self.state_manager.create_workflow_session(
            workflow_type, user_id, payload
        )
        
        try:
            # 2. 设置进度回调
            async def progress_callback(message: str):
                await status_broadcaster.broadcast_workflow_progress(session_id, {
                    "message": message,
                    "timestamp": datetime.now().isoformat()
                })
            
            # 3. 更新状态为运行中
            await self.state_manager.update_workflow_status(
                session_id, WorkflowStatus.RUNNING
            )
            
            # 4. 执行HTTP请求
            result = await http_enhancer.enhanced_request(
                url, payload, session_id, progress_callback
            )
            
            # 5. 更新完成状态
            final_status = WorkflowStatus.COMPLETED if result["success"] else WorkflowStatus.FAILED
            await self.state_manager.update_workflow_status(
                session_id, final_status, result, result.get("message")
            )
            
            return {
                "session_id": session_id,
                "result": result,
                "status": final_status.value
            }
            
        except Exception as e:
            # 更新失败状态
            await self.state_manager.update_workflow_status(
                session_id, WorkflowStatus.FAILED, error_info=str(e)
            )
            
            # 广播错误信息
            await status_broadcaster.broadcast_workflow_error(session_id, {
                "error": str(e),
                "timestamp": datetime.now().isoformat()
            })
            
            raise
    
    async def _on_status_change(self, session_id: str, status: WorkflowStatus, workflow_data: Dict[str, Any]):
        """状态变更回调"""
        
        # 通过WebSocket广播状态变更
        await status_broadcaster.broadcast_workflow_status(session_id, {
            "status": status.value,
            "workflow_data": workflow_data,
            "timestamp": datetime.now().isoformat()
        })
