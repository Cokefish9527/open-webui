## 1. 系统概述

### 1.1 设计目标

构建一个实时状态反馈系统，为用户提供工作流执行过程的实时进度、状态更新和结果反馈，提升用户体验和系统透明度。

### 1.2 核心功能

- **实时进度显示**：工作流执行进度的实时更新
- **状态变更通知**：工作流状态变化的即时通知
- **错误信息反馈**：详细的错误信息和解决建议
- **结果预览**：部分结果的实时预览功能

```python
# backend/open_webui/realtime/status_manager.py
import asyncio
import json
from typing import Dict, Any, List, Optional, Callable
from datetime import datetime
from enum import Enum
from dataclasses import dataclass, asdict

class WorkflowStatus(Enum):
    """工作流状态枚举"""
    PENDING = "pending"
    INITIALIZING = "initializing"
    RUNNING = "running"
    PROCESSING = "processing"
    COMPLETING = "completing"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

class StatusType(Enum):
    """状态类型"""
    PROGRESS = "progress"
    STATUS_CHANGE = "status_change"
    ERROR = "error"
    WARNING = "warning"
    INFO = "info"
    RESULT_PREVIEW = "result_preview"

@dataclass
class StatusUpdate:
    """状态更新数据结构"""
    session_id: str
    status_type: StatusType
    workflow_status: WorkflowStatus
    progress: float  # 0.0 - 100.0
    message: str
    details: Dict[str, Any]
    timestamp: datetime
    
    def to_dict(self) -> Dict[str, Any]:
        return {
            "session_id": self.session_id,
            "status_type": self.status_type.value,
            "workflow_status": self.workflow_status.value,
            "progress": self.progress,
            "message": self.message,
            "details": self.details,
            "timestamp": self.timestamp.isoformat()
        }

class StatusManager:
    """状态管理器"""
    
    def __init__(self):
        self.active_sessions: Dict[str, Dict[str, Any]] = {}
        self.status_history: Dict[str, List[StatusUpdate]] = {}
        self.subscribers: Dict[str, List[Callable]] = {}
        self.global_subscribers: List[Callable] = []
    
    async def create_session(self, session_id: str, workflow_type: str, user_id: str) -> None:
        """创建新的状态会话"""
        
        self.active_sessions[session_id] = {
            "workflow_type": workflow_type,
            "user_id": user_id,
            "status": WorkflowStatus.PENDING,
            "progress": 0.0,
            "created_at": datetime.now(),
            "last_update": datetime.now()
        }
        
        self.status_history[session_id] = []
        
        # 发送初始状态
        await self.update_status(
            session_id,
            StatusType.STATUS_CHANGE,
            WorkflowStatus.PENDING,
            0.0,
            "工作流已创建，等待执行",
            {"workflow_type": workflow_type}
        )
    
    async def update_status(self, 
                          session_id: str,
                          status_type: StatusType,
                          workflow_status: WorkflowStatus,
                          progress: float,
                          message: str,
                          details: Dict[str, Any] = None) -> None:
        """更新状态"""
        
        if session_id not in self.active_sessions:
            return
        
        # 更新会话信息
        self.active_sessions[session_id].update({
            "status": workflow_status,
            "progress": progress,
            "last_update": datetime.now()
        })
        
        # 创建状态更新对象
        status_update = StatusUpdate(
            session_id=session_id,
            status_type=status_type,
            workflow_status=workflow_status,
            progress=progress,
            message=message,
            details=details or {},
            timestamp=datetime.now()
        )
        
        # 添加到历史记录
        self.status_history[session_id].append(status_update)
        
        # 通知订阅者
        await self._notify_subscribers(session_id, status_update)
    
    async def update_progress(self, session_id: str, progress: float, message: str = None) -> None:
        """更新进度"""
        
        if session_id not in self.active_sessions:
            return
        
        current_status = self.active_sessions[session_id]["status"]
        
        await self.update_status(
            session_id,
            StatusType.PROGRESS,
            current_status,
            progress,
            message or f"进度: {progress:.1f}%",
            {"progress_only": True}
        )
    
    async def add_result_preview(self, session_id: str, preview_data: Dict[str, Any]) -> None:
        """添加结果预览"""
        
        if session_id not in self.active_sessions:
            return
        
        current_status = self.active_sessions[session_id]["status"]
        current_progress = self.active_sessions[session_id]["progress"]
        
        await self.update_status(
            session_id,
            StatusType.RESULT_PREVIEW,
            current_status,
            current_progress,
            "部分结果已生成",
            {"preview": preview_data}
        )
    
    async def report_error(self, session_id: str, error_message: str, error_details: Dict[str, Any] = None) -> None:
        """报告错误"""
        
        await self.update_status(
            session_id,
            StatusType.ERROR,
            WorkflowStatus.FAILED,
            self.active_sessions.get(session_id, {}).get("progress", 0.0),
            error_message,
            {"error": True, "error_details": error_details or {}}
        )
    
    def subscribe(self, session_id: str, callback: Callable) -> None:
        """订阅特定会话的状态更新"""
        
        if session_id not in self.subscribers:
            self.subscribers[session_id] = []
        
        self.subscribers[session_id].append(callback)
    
    def subscribe_global(self, callback: Callable) -> None:
        """订阅全局状态更新"""
        self.global_subscribers.append(callback)
    
    async def _notify_subscribers(self, session_id: str, status_update: StatusUpdate) -> None:
        """通知订阅者"""
        
        # 通知特定会话订阅者
        if session_id in self.subscribers:
            for callback in self.subscribers[session_id]:
                try:
                    await callback(status_update)
                except Exception as e:
                    logging.error(f"Status subscriber callback failed: {e}")
        
        # 通知全局订阅者
        for callback in self.global_subscribers:
            try:
                await callback(status_update)
            except Exception as e:
                logging.error(f"Global status subscriber callback failed: {e}")
    
    def get_session_status(self, session_id: str) -> Optional[Dict[str, Any]]:
        """获取会话状态"""
        return self.active_sessions.get(session_id)
    
    def get_session_history(self, session_id: str) -> List[Dict[str, Any]]:
        """获取会话历史"""
        
        if session_id not in self.status_history:
            return []
        
        return [update.to_dict() for update in self.status_history[session_id]]
    
    async def close_session(self, session_id: str) -> None:
        """关闭会话"""
        
        if session_id in self.active_sessions:
            del self.active_sessions[session_id]
        
        if session_id in self.subscribers:
            del self.subscribers[session_id]
        
        # 保留历史记录一段时间
        # 可以考虑定期清理

# 全局状态管理器
status_manager = StatusManager()
```

```python
# backend/open_webui/realtime/message_router.py
import json
import logging
from typing import Dict, Any, Set
from .status_manager import status_manager, StatusUpdate

class MessageRouter:
    """WebSocket消息路由器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.connections: Dict[str, Set[str]] = {}  # user_id -> session_ids
        self.session_connections: Dict[str, str] = {}  # session_id -> user_id
        
        # 订阅状态管理器的全局更新
        status_manager.subscribe_global(self._handle_status_update)
    
    def register_connection(self, user_id: str, session_id: str) -> None:
        """注册WebSocket连接"""
        
        if user_id not in self.connections:
            self.connections[user_id] = set()
        
        self.connections[user_id].add(session_id)
        self.session_connections[session_id] = user_id
        
        self.logger.info(f"WebSocket connection registered: user={user_id}, session={session_id}")
    
    def unregister_connection(self, session_id: str) -> None:
        """注销WebSocket连接"""
        
        if session_id in self.session_connections:
            user_id = self.session_connections[session_id]
            
            if user_id in self.connections:
                self.connections[user_id].discard(session_id)
                
                if not self.connections[user_id]:
                    del self.connections[user_id]
            
            del self.session_connections[session_id]
            
            self.logger.info(f"WebSocket connection unregistered: session={session_id}")
    
    async def _handle_status_update(self, status_update: StatusUpdate) -> None:
        """处理状态更新"""
        
        session_id = status_update.session_id
        
        if session_id in self.session_connections:
            user_id = self.session_connections[session_id]
            
            # 构造WebSocket消息
            message = {
                "type": "status_update",
                "data": status_update.to_dict()
            }
            
            # 发送给对应用户的所有连接
            await self._send_to_user(user_id, message)
    
    async def _send_to_user(self, user_id: str, message: Dict[str, Any]) -> None:
        """发送消息给用户"""
        
        if user_id not in self.connections:
            return
        
        # 这里需要与WebSocket服务器集成
        # 实际实现中会调用WebSocket管理器的发送方法
        from ..websocket.client_manager import ws_client_manager
        
        sent_count = await ws_client_manager.send_to_user(user_id, message)
        
        if sent_count == 0:
            self.logger.warning(f"Failed to send message to user {user_id}")
```

# backend/open_webui/realtime/progress_tracker.py
import asyncio
from typing import Dict, Any, List, Optional
from datetime import datetime, timedelta
from .status_manager import status_manager, WorkflowStatus, StatusType

class ProgressStage:
    """进度阶段定义"""
    
    def __init__(self, name: str, weight: float, description: str):
        self.name = name
        self.weight = weight  # 权重 (0.0 - 1.0)
        self.description = description
        self.completed = False
        self.progress = 0.0

class ProgressTracker:
    """进度跟踪器"""
    
    def __init__(self):
        self.workflow_stages = self._initialize_workflow_stages()
        self.active_trackers: Dict[str, Dict[str, Any]] = {}
    
    def _initialize_workflow_stages(self) -> Dict[str, List[ProgressStage]]:
        """初始化工作流阶段定义"""
        
        return {
            "main_workflow": [
                ProgressStage("初始化", 0.1, "准备AI模型和工具"),
                ProgressStage("输入处理", 0.2, "分析用户输入"),
                ProgressStage("工具调用", 0.5, "执行AI工具"),
                ProgressStage("结果生成", 0.15, "生成响应内容"),
                ProgressStage("完成", 0.05, "最终处理")
            ],
            "company_info": [
                ProgressStage("文件解析", 0.3, "解析上传的文件"),
                ProgressStage("信息提取", 0.4, "提取企业信息"),
                ProgressStage("关键词分析", 0.2, "生成关键词"),
                ProgressStage("结果整理", 0.1, "整理分析结果")
            ],
            "video_scraping": [
                ProgressStage("爬虫启动", 0.1, "启动视频爬虫"),
                ProgressStage("数据抓取", 0.4, "抓取视频数据"),
                ProgressStage("内容分析", 0.4, "分析视频内容"),
                ProgressStage("结果汇总", 0.1, "汇总分析结果")
            ],
            "viral_learning": [
                ProgressStage("数据查询", 0.2, "查询历史数据"),
                ProgressStage("趋势分析", 0.5, "分析爆款趋势"),
                ProgressStage("学习更新", 0.2, "更新学习模型"),
                ProgressStage("结果生成", 0.1, "生成学习报告")
            ]
        }
    
    async def start_tracking(self, session_id: str, workflow_type: str) -> None:
        """开始进度跟踪"""
        
        if workflow_type not in self.workflow_stages:
            workflow_type = "main_workflow"  # 默认工作流
        
        stages = [
            ProgressStage(stage.name, stage.weight, stage.description)
            for stage in self.workflow_stages[workflow_type]
        ]
        
        self.active_trackers[session_id] = {
            "workflow_type": workflow_type,
            "stages": stages,
            "current_stage": 0,
            "overall_progress": 0.0,
            "start_time": datetime.now()
        }
        
        # 更新初始状态
        await status_manager.update_status(
            session_id,
            StatusType.STATUS_CHANGE,
            WorkflowStatus.INITIALIZING,
            0.0,
            f"开始执行{workflow_type}工作流",
            {"stages": [{"name": s.name, "description": s.description} for s in stages]}
        )
    
    async def update_stage_progress(self, session_id: str, stage_progress: float, message: str = None) -> None:
        """更新当前阶段进度"""
        
        if session_id not in self.active_trackers:
            return
        
        tracker = self.active_trackers[session_id]
        stages = tracker["stages"]
        current_stage_idx = tracker["current_stage"]
        
        if current_stage_idx >= len(stages):
            return
        
        current_stage = stages[current_stage_idx]
        current_stage.progress = min(stage_progress, 100.0)
        
        # 计算总体进度
        overall_progress = self._calculate_overall_progress(stages, current_stage_idx)
        tracker["overall_progress"] = overall_progress
        
        # 更新状态
        stage_message = message or f"{current_stage.description} - {stage_progress:.1f}%"
        
        await status_manager.update_progress(
            session_id,
            overall_progress,
            stage_message
        )
    
    async def complete_current_stage(self, session_id: str, message: str = None) -> None:
        """完成当前阶段"""
        
        if session_id not in self.active_trackers:
            return
        
        tracker = self.active_trackers[session_id]
        stages = tracker["stages"]
        current_stage_idx = tracker["current_stage"]
        
        if current_stage_idx >= len(stages):
            return
        
        # 标记当前阶段完成
        current_stage = stages[current_stage_idx]
        current_stage.completed = True
        current_stage.progress = 100.0
        
        # 移动到下一阶段
        tracker["current_stage"] = current_stage_idx + 1
        
        # 计算总体进度
        overall_progress = self._calculate_overall_progress(stages, current_stage_idx + 1)
        tracker["overall_progress"] = overall_progress
        
        # 检查是否全部完成
        if current_stage_idx + 1 >= len(stages):
            await status_manager.update_status(
                session_id,
                StatusType.STATUS_CHANGE,
                WorkflowStatus.COMPLETED,
                100.0,
                "工作流执行完成",
                {"completed_at": datetime.now().isoformat()}
            )
        else:
            next_stage = stages[current_stage_idx + 1]
            stage_message = message or f"开始{next_stage.description}"
            
            await status_manager.update_status(
                session_id,
                StatusType.PROGRESS,
                WorkflowStatus.RUNNING,
                overall_progress,
                stage_message,
                {
                    "current_stage": next_stage.name,
                    "stage_description": next_stage.description
                }
            )
    
    def _calculate_overall_progress(self, stages: List[ProgressStage], current_stage_idx: int) -> float:
        """计算总体进度"""
        
        total_progress = 0.0
        
        # 已完成阶段的进度
        for i in range(min(current_stage_idx, len(stages))):
            total_progress += stages[i].weight * 100.0
        
        # 当前阶段的进度
        if current_stage_idx < len(stages):
            current_stage = stages[current_stage_idx]
            total_progress += current_stage.weight * current_stage.progress
        
        return min(total_progress, 100.0)
    
    async def report_stage_error(self, session_id: str, error_message: str) -> None:
        """报告阶段错误"""
        
        if session_id not in self.active_trackers:
            return
        
        tracker = self.active_trackers[session_id]
        current_stage_idx = tracker["current_stage"]
        
        stage_name = "未知阶段"
        if current_stage_idx < len(tracker["stages"]):
            stage_name = tracker["stages"][current_stage_idx].name
        
        await status_manager.report_error(
            session_id,
            f"{stage_name}执行失败: {error_message}",
            {
                "stage": stage_name,
                "stage_index": current_stage_idx,
                "error_type": "stage_error"
            }
        )
    
    def get_tracker_info(self, session_id: str) -> Optional[Dict[str, Any]]:
        """获取跟踪器信息"""
        
        if session_id not in self.active_trackers:
            return None
        
        tracker = self.active_trackers[session_id]
        stages = tracker["stages"]
        
        return {
            "workflow_type": tracker["workflow_type"],
            "overall_progress": tracker["overall_progress"],
            "current_stage": tracker["current_stage"],
            "stages": [
                {
                    "name": stage.name,
                    "description": stage.description,
                    "weight": stage.weight,
                    "progress": stage.progress,
                    "completed": stage.completed
                }
                for stage in stages
            ],
            "start_time": tracker["start_time"].isoformat(),
            "elapsed_time": (datetime.now() - tracker["start_time"]).total_seconds()
        }

# 全局进度跟踪器
progress_tracker = ProgressTracker()

// src/lib/components/RealtimeStatus.svelte
<script>
    import { onMount, onDestroy } from 'svelte';
    import { workflowSocketClient } from '../websocket/WorkflowSocketClient.js';
    
    export let sessionId;
    export let workflowType;
    
    let status = {
        workflow_status: 'pending',
        progress: 0,
        message: '准备中...',
        current_stage: null,
        stages: []
    };
    
    let notifications = [];
    let showDetails = false;
    
    onMount(() => {
        // 订阅状态更新
        workflowSocketClient.on('status_update', handleStatusUpdate);
        
        // 订阅工作流
        if (sessionId) {
            workflowSocketClient.subscribeToWorkflow(sessionId);
        }
    });
    
    onDestroy(() => {
        workflowSocketClient.off('status_update', handleStatusUpdate);
        
        if (sessionId) {
            workflowSocketClient.unsubscribeFromWorkflow(sessionId);
        }
    });
    
    function handleStatusUpdate(data) {
        if (data.session_id !== sessionId) return;
        
        const update = data.data;
        
        // 更新状态
        status = {
            ...status,
            workflow_status: update.workflow_status,
            progress: update.progress,
            message: update.message
        };
        
        // 处理不同类型的更新
        switch (update.status_type) {
            case 'progress':
                updateProgress(update);
                break;
            case 'status_change':
                updateStatus(update);
                break;
            case 'error':
                handleError(update);
                break;
            case 'result_preview':
                handleResultPreview(update);
                break;
        }
        
        // 添加通知
        addNotification(update);
    }
    
    function updateProgress(update) {
        if (update.details.current_stage) {
            status.current_stage = update.details.current_stage;
        }
        
        if (update.details.stages) {
            status.stages = update.details.stages;
        }
    }
    
    function updateStatus(update) {
        if (update.details.stages) {
            status.stages = update.details.stages;
        }
    }
    
    function handleError(update) {
        notifications = [...notifications, {
            type: 'error',
            message: update.message,
            timestamp: new Date(update.timestamp),
            details: update.details
        }];
    }
    
    function handleResultPreview(update) {
        if (update.details.preview) {
            // 处理结果预览
            notifications = [...notifications, {
                type: 'preview',
                message: '部分结果已生成',
                timestamp: new Date(update.timestamp),
                preview: update.details.preview
            }];
        }
    }
    
    function addNotification(update) {
        const notification = {
            type: update.status_type,
            message: update.message,
            timestamp: new Date(update.timestamp)
        };
        
        notifications = [notification, ...notifications].slice(0, 10); // 保留最近10条
    }
    
    function getStatusColor(status) {
        const colors = {
            'pending': 'text-yellow-600',
            'initializing': 'text-blue-600',
            'running': 'text-blue-600',
            'processing': 'text-blue-600',
            'completing': 'text-green-600',
            'completed': 'text-green-600',
            'failed': 'text-red-600',
            'cancelled': 'text-gray-600'
        };
        return colors[status] || 'text-gray-600';
    }
    
    function getStatusIcon(status) {
        const icons = {
            'pending': '⏳',
            'initializing': '🔄',
            'running': '⚡',
            'processing': '🔄',
            'completing': '✅',
            'completed': '✅',
            'failed': '❌',
            'cancelled': '⏹️'
        };
        return icons[status] || '❓';
    }
</script>

<div class="realtime-status bg-white rounded-lg shadow-md p-6">
    <!-- 状态头部 -->
    <div class="flex items-center justify-between mb-4">
        <div class="flex items-center space-x-3">
            <span class="text-2xl">{getStatusIcon(status.workflow_status)}</span>
            <div>
                <h3 class="text-lg font-semibold {getStatusColor(status.workflow_status)}">
                    {status.workflow_status.toUpperCase()}
                </h3>
                <p class="text-sm text-gray-600">{workflowType}</p>
            </div>
        </div>
        
        <button 
            class="text-sm text-blue-600 hover:text-blue-800"
            on:click={() => showDetails = !showDetails}
        >
            {showDetails ? '隐藏详情' : '显示详情'}
        </button>
    </div>
    
    <!-- 进度条 -->
    <div class="mb-4">
        <div class="flex justify-between text-sm text-gray-600 mb-1">
            <span>{status.message}</span>
            <span>{status.progress.toFixed(1)}%</span>
        </div>
        <div class="w-full bg-gray-200 rounded-full h-2">
            <div 
                class="bg-blue-600 h-2 rounded-full transition-all duration-300"
                style="width: {status.progress}%"
            ></div>
        </div>
    </div>
    
    <!-- 阶段详情 -->
    {#if showDetails && status.stages.length > 0}
        <div class="mb-4">
            <h4 class="text-sm font-medium text-gray-700 mb-2">执行阶段</h4>
            <div class="space-y-2">
                {#each status.stages as stage, index}
                    <div class="flex items-center space-x-3 text-sm">
                        <span class="w-6 h-6 rounded-full flex items-center justify-center text-xs
                                   {stage.completed ? 'bg-green-100 text-green-600' : 
                                    status.current_stage === stage.name ? 'bg-blue-100 text-blue-600' : 
                                    'bg-gray-100 text-gray-400'}">
                            {stage.completed ? '✓' : index + 1}
                        </span>
                        <div class="flex-1">
                            <div class="font-medium">{stage.name}</div>
                            <div class="text-gray-500">{stage.description}</div>
                        </div>
                        {#if stage.progress > 0}
                            <span class="text-xs text-gray-500">{stage.progress.toFixed(0)}%</span>
                        {/if}
                    </div>
                {/each}
            </div>
        </div>
    {/if}
    
    <!-- 通知列表 -->
    {#if showDetails && notifications.length > 0}
        <div>
            <h4 class="text-sm font-medium text-gray-700 mb-2">最近通知</h4>
            <div class="space-y-2 max-h-40 overflow-y-auto">
                {#each notifications as notification}
                    <div class="text-xs p-2 rounded
                               {notification.type === 'error' ? 'bg-red-50 text-red-700' :
                                notification.type === 'preview' ? 'bg-green-50 text-green-700' :
                                'bg-gray-50 text-gray-700'}">
                        <div class="font-medium">{notification.message}</div>
                        <div class="text-gray-500">
                            {notification.timestamp.toLocaleTimeString()}
                        </div>
                    </div>
                {/each}
            </div>
        </div>
    {/if}
</div>

        min-height: 200px;
    }
