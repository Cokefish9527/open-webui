## 1. 设计目标

构建一个全面的错误处理与容错系统，确保OpenWebUI与n8n工作流集成的高可用性和稳定性。

### 1.2 核心原则

- **故障隔离**：单个组件故障不影响整体系统
- **快速恢复**：自动检测和恢复机制
- **优雅降级**：在部分功能不可用时提供基础服务
- **透明处理**：用户感知最小化的错误处理

### 2.2 错误处理策略

```python
# backend/open_webui/utils/error_handler.py
import logging
import asyncio
from typing import Dict, Any, Optional, Callable
from datetime import datetime, timedelta
from .error_types import ErrorCategory, ErrorSeverity, ErrorCode

class ErrorHandler:
    """统一错误处理器"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.error_strategies = self._initialize_strategies()
        self.circuit_breakers = {}
        self.retry_configs = self._initialize_retry_configs()
    
    def _initialize_strategies(self) -> Dict[ErrorCode, Callable]:
        """初始化错误处理策略"""
        return {
            ErrorCode.CONNECTION_TIMEOUT: self._handle_connection_timeout,
            ErrorCode.WORKFLOW_EXECUTION_FAILED: self._handle_workflow_failure,
            ErrorCode.INVALID_JSON_FORMAT: self._handle_json_format_error,
            ErrorCode.SERVICE_UNAVAILABLE: self._handle_service_unavailable,
        }
    
    def _initialize_retry_configs(self) -> Dict[ErrorCode, Dict[str, Any]]:
        """初始化重试配置"""
        return {
            ErrorCode.CONNECTION_TIMEOUT: {
                "max_retries": 3,
                "base_delay": 1.0,
                "max_delay": 30.0,
                "backoff_factor": 2.0
            },
            ErrorCode.WORKFLOW_EXECUTION_FAILED: {
                "max_retries": 2,
                "base_delay": 2.0,
                "max_delay": 60.0,
                "backoff_factor": 3.0
            },
            ErrorCode.SERVICE_UNAVAILABLE: {
                "max_retries": 5,
                "base_delay": 0.5,
                "max_delay": 10.0,
                "backoff_factor": 1.5
            }
        }
    
    async def handle_error(self, error: Exception, context: Dict[str, Any] = None) -> Dict[str, Any]:
        """统一错误处理入口"""
        
        # 1. 错误分析和分类
        error_info = self._analyze_error(error, context)
        
        # 2. 记录错误
        self._log_error(error_info)
        
        # 3. 执行处理策略
        handling_result = await self._execute_handling_strategy(error_info)
        
        # 4. 更新熔断器状态
        self._update_circuit_breaker(error_info)
        
        return handling_result
    
    def _analyze_error(self, error: Exception, context: Dict[str, Any] = None) -> Dict[str, Any]:
        """分析错误并分类"""
        
        error_type = type(error).__name__
        error_message = str(error)
        
        # 根据错误类型和消息确定错误代码
        error_code = self._determine_error_code(error_type, error_message)
        
        # 确定错误类别和严重程度
        category = self._determine_category(error_code)
        severity = self._determine_severity(error_code, context)
        
        return {
            "error_code": error_code,
            "category": category,
            "severity": severity,
            "error_type": error_type,
            "message": error_message,
            "context": context or {},
            "timestamp": datetime.now(),
            "traceback": self._get_traceback(error)
        }
    
    async def _execute_handling_strategy(self, error_info: Dict[str, Any]) -> Dict[str, Any]:
        """执行错误处理策略"""
        
        error_code = error_info["error_code"]
        
        if error_code in self.error_strategies:
            try:
                return await self.error_strategies[error_code](error_info)
            except Exception as e:
                self.logger.error(f"Error handling strategy failed: {e}")
                return self._create_fallback_response(error_info)
        else:
            return self._create_default_response(error_info)
    
    async def _handle_connection_timeout(self, error_info: Dict[str, Any]) -> Dict[str, Any]:
        """处理连接超时错误"""
        
        context = error_info["context"]
        retry_count = context.get("retry_count", 0)
        max_retries = self.retry_configs[ErrorCode.CONNECTION_TIMEOUT]["max_retries"]
        
        if retry_count < max_retries:
            # 计算延迟时间
            delay = self._calculate_retry_delay(ErrorCode.CONNECTION_TIMEOUT, retry_count)
            
            return {
                "success": False,
                "error_code": ErrorCode.CONNECTION_TIMEOUT.value,
                "message": f"连接超时，将在 {delay} 秒后重试",
                "retry_after": delay,
                "retry_count": retry_count + 1,
                "recoverable": True
            }
        else:
            return {
                "success": False,
                "error_code": ErrorCode.CONNECTION_TIMEOUT.value,
                "message": "连接超时，已达到最大重试次数",
                "recoverable": False,
                "fallback_available": True
            }
    
    async def _handle_workflow_failure(self, error_info: Dict[str, Any]) -> Dict[str, Any]:
        """处理工作流执行失败"""
        
        context = error_info["context"]
        workflow_type = context.get("workflow_type")
        
        # 检查是否有备用工作流
        fallback_workflow = self._get_fallback_workflow(workflow_type)
        
        if fallback_workflow:
            return {
                "success": False,
                "error_code": ErrorCode.WORKFLOW_EXECUTION_FAILED.value,
                "message": "主工作流执行失败，切换到备用工作流",
                "fallback_workflow": fallback_workflow,
                "recoverable": True
            }
        else:
            return {
                "success": False,
                "error_code": ErrorCode.WORKFLOW_EXECUTION_FAILED.value,
                "message": "工作流执行失败，无可用备用方案",
                "recoverable": False,
                "suggested_action": "请稍后重试或联系技术支持"
            }
```

## 3. 熔断器机制

### 3.1 熔断器实现

```python
# backend/open_webui/utils/circuit_breaker.py
import asyncio
from datetime import datetime, timedelta
from enum import Enum
from typing import Dict, Any, Callable

class CircuitState(Enum):
    CLOSED = "closed"      # 正常状态
    OPEN = "open"          # 熔断状态
    HALF_OPEN = "half_open"  # 半开状态

class CircuitBreaker:
    """熔断器实现"""
    
    def __init__(self, 
                 failure_threshold: int = 5,
                 recovery_timeout: int = 60,
                 expected_exception: Exception = Exception):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.expected_exception = expected_exception
        
        self.failure_count = 0
        self.last_failure_time = None
        self.state = CircuitState.CLOSED
    
    async def call(self, func: Callable, *args, **kwargs):
        """通过熔断器调用函数"""
        
        if self.state == CircuitState.OPEN:
            if self._should_attempt_reset():
                self.state = CircuitState.HALF_OPEN
            else:
                raise Exception("Circuit breaker is OPEN")
        
        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        
        except self.expected_exception as e:
            self._on_failure()
            raise e
    
    def _should_attempt_reset(self) -> bool:
        """检查是否应该尝试重置熔断器"""
        return (
            self.last_failure_time and
            datetime.now() - self.last_failure_time >= timedelta(seconds=self.recovery_timeout)
        )
    
    def _on_success(self):
        """成功调用时的处理"""
        self.failure_count = 0
        self.state = CircuitState.CLOSED
    
    def _on_failure(self):
        """失败调用时的处理"""
        self.failure_count += 1
        self.last_failure_time = datetime.now()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

class CircuitBreakerManager:
    """熔断器管理器"""
    
    def __init__(self):
        self.breakers: Dict[str, CircuitBreaker] = {}
    
    def get_breaker(self, name: str, **kwargs) -> CircuitBreaker:
        """获取或创建熔断器"""
        if name not in self.breakers:
            self.breakers[name] = CircuitBreaker(**kwargs)
        return self.breakers[name]
    
    def get_status(self) -> Dict[str, Dict[str, Any]]:
        """获取所有熔断器状态"""
        status = {}
        for name, breaker in self.breakers.items():
            status[name] = {
                "state": breaker.state.value,
                "failure_count": breaker.failure_count,
                "last_failure_time": breaker.last_failure_time.isoformat() if breaker.last_failure_time else None
            }
        return status

# 全局熔断器管理器
circuit_breaker_manager = CircuitBreakerManager()
```

# backend/open_webui/utils/fallback_service.py
from typing import Dict, Any, Optional
import logging

class FallbackService:
    """服务降级处理"""
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.fallback_strategies = self._initialize_fallback_strategies()
    
    def _initialize_fallback_strategies(self) -> Dict[str, Callable]:
        """初始化降级策略"""
        return {
            "main_workflow": self._fallback_main_workflow,
            "company_info": self._fallback_company_info,
            "video_scraping": self._fallback_video_scraping,
            "viral_learning": self._fallback_viral_learning
        }
    
    async def execute_fallback(self, service_name: str, original_request: Dict[str, Any]) -> Dict[str, Any]:
        """执行服务降级"""
        
        if service_name in self.fallback_strategies:
            try:
                return await self.fallback_strategies[service_name](original_request)
            except Exception as e:
                self.logger.error(f"Fallback strategy failed for {service_name}: {e}")
                return self._create_minimal_response(original_request)
        else:
            return self._create_minimal_response(original_request)
    
    async def _fallback_main_workflow(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """主工作流降级处理"""
        
        user_input = request.get("input", "")
        
        # 使用本地LLM或简单规则生成响应
        fallback_response = {
            "content": f"抱歉，当前AI服务暂时不可用。您的问题是：{user_input}。请稍后重试或联系技术支持。",
            "type": "fallback_response",
            "metadata": {
                "fallback": True,
                "original_service": "main_workflow",
                "timestamp": datetime.now().isoformat()
            }
        }
        
        return {
            "success": True,
            "data": fallback_response,
            "fallback": True,
            "message": "使用降级服务处理请求"
        }
    
    async def _fallback_company_info(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """企业信息收集降级处理"""
        
        return {
            "success": True,
            "data": {
                "company_name": "未知企业",
                "keywords": ["企业分析", "信息收集"],
                "analysis": {
                    "status": "服务暂时不可用",
                    "message": "企业信息分析服务正在维护中，请稍后重试"
                }
            },
            "fallback": True,
            "message": "企业信息分析服务暂时不可用"
        }
    
    def _create_minimal_response(self, request: Dict[str, Any]) -> Dict[str, Any]:
        """创建最小化响应"""
        
        return {
            "success": False,
            "data": {},
            "message": "服务暂时不可用，请稍后重试",
            "fallback": True,
            "error_code": "SERVICE_UNAVAILABLE"
        }

# 全局降级服务
fallback_service = FallbackService()

## 5. 监控与告警

### 5.1 错误监控

```python
# backend/open_webui/utils/error_monitor.py
import asyncio
from collections import defaultdict, deque
from datetime import datetime, timedelta
from typing import Dict, Any, List

class ErrorMonitor:
    """错误监控器"""
    
    def __init__(self):
        self.error_counts = defaultdict(int)
        self.error_history = deque(maxlen=1000)
        self.alert_thresholds = {
            "error_rate": 0.1,  # 10%错误率
            "error_count": 50,   # 50个错误/小时
            "critical_errors": 5  # 5个严重错误
        }
        self.alert_callbacks = []
    
    def record_error(self, error_info: Dict[str, Any]):
        """记录错误"""
        
        error_key = f"{error_info['category'].value}_{error_info['error_code'].value}"
        self.error_counts[error_key] += 1
        
        self.error_history.append({
            "timestamp": datetime.now(),
            "error_info": error_info
        })
        
        # 检查是否需要告警
        asyncio.create_task(self._check_alerts(error_info))
    
    async def _check_alerts(self, error_info: Dict[str, Any]):
        """检查告警条件"""
        
        # 检查错误率
        error_rate = self._calculate_error_rate()
        if error_rate > self.alert_thresholds["error_rate"]:
            await self._trigger_alert("high_error_rate", {
                "error_rate": error_rate,
                "threshold": self.alert_thresholds["error_rate"]
            })
        
        # 检查严重错误
        if error_info["severity"] == ErrorSeverity.CRITICAL:
            critical_count = self._count_recent_critical_errors()
            if critical_count >= self.alert_thresholds["critical_errors"]:
                await self._trigger_alert("critical_errors", {
                    "count": critical_count,
                    "threshold": self.alert_thresholds["critical_errors"]
                })
    
    def _calculate_error_rate(self) -> float:
        """计算错误率"""
        
        now = datetime.now()
        hour_ago = now - timedelta(hours=1)
        
        recent_errors = [
            entry for entry in self.error_history
            if entry["timestamp"] > hour_ago
        ]
        
        if not recent_errors:
            return 0.0
        
        # 这里需要总请求数来计算错误率
        # 简化实现，假设每个错误对应10个正常请求
        total_requests = len(recent_errors) * 10
        return len(recent_errors) / total_requests if total_requests > 0 else 0.0
    
    async def _trigger_alert(self, alert_type: str, data: Dict[str, Any]):
        """触发告警"""
        
        alert_info = {
            "type": alert_type,
            "timestamp": datetime.now(),
            "data": data,
            "severity": "high" if alert_type == "critical_errors" else "medium"
        }
        
        for callback in self.alert_callbacks:
            try:
                await callback(alert_info)
            except Exception as e:
                logging.error(f"Alert callback failed: {e}")
    
    def add_alert_callback(self, callback: Callable):
        """添加告警回调"""
        self.alert_callbacks.append(callback)
    
    def get_error_statistics(self) -> Dict[str, Any]:
        """获取错误统计"""
        
        now = datetime.now()
        hour_ago = now - timedelta(hours=1)
        day_ago = now - timedelta(days=1)
        
        recent_errors = [
            entry for entry in self.error_history
            if entry["timestamp"] > hour_ago
        ]
        
        daily_errors = [
            entry for entry in self.error_history
            if entry["timestamp"] > day_ago
        ]
        
        return {
            "total_errors": len(self.error_history),
            "recent_errors": len(recent_errors),
            "daily_errors": len(daily_errors),
            "error_rate": self._calculate_error_rate(),
            "top_errors": self._get_top_errors(),
            "timestamp": now.isoformat()
        }

# 全局错误监控器
error_monitor = ErrorMonitor()
```

这个错误处理与容错系统提供了：

1. **全面的错误分类**：系统化的错误类型和处理策略

3. **熔断器保护**：防止级联故障的熔断机制
4. **服务降级**：在服务不可用时提供基础功能
5. **实时监控**：错误统计和告警机制
6. **自动恢复**：故障自动检测和恢复能力

通过这个系统，可以大大提高OpenWebUI与n8n集成的稳定性和可靠性。
